   152→  private val onBackPressedCallback: OnBackPressedCallback = object : OnBackPressedCallback(true) {
   153→    override fun handleOnBackPressed() {
   154→      if (binding.root.isDrawerOpen(GravityCompat.START)) {
   155→        binding.root.closeDrawer(GravityCompat.START)
   156→      } else if (editorBottomSheet?.state != BottomSheetBehavior.STATE_COLLAPSED) {

   158→      } else if (binding.swipeReveal.isOpen) {

   161→        doConfirmProjectClose()
   162→      }
   163→    }

   166→  private val memoryUsageListener = MemoryUsageWatcher.MemoryUsageListener { memoryUsage ->
   167→    memoryUsage.forEachValue { proc ->
   168→      _binding?.memUsageView?.chart?.apply {
   169→        val dataset = (data.getDataSetByIndex(pidToDatasetIdxMap[proc.pid]) as LineDataSet?)
   170→          ?: run {
   171→            log.error("No dataset found for process: {}: {}", proc.pid, proc.pname)
   172→            return@forEachValue
   173→          }
   174→
   175→        dataset.entries.mapIndexed { index, entry ->
   176→          entry.y = byte2MemorySize(proc.usageHistory[index], MemoryConstants.MB).toFloat()
   177→        }
   178→
   179→        dataset.label = "%s - %.2fMB".format(proc.pname, dataset.entries.last().y)
   180→        dataset.notifyDataSetChanged()
   181→        data.notifyDataChanged()
   182→        notifyDataSetChanged()
   183→        invalidate()
   184→      }
   185→    }
   186→  }

   232→  protected open fun preDestroy() {
   233→    _binding = null
   234→
   235→    optionsMenuInvalidator?.also {
   236→      ThreadUtils.getMainHandler().removeCallbacks(it)
   237→    }
   238→
   239→    optionsMenuInvalidator = null
   240→

   244→    if (isDestroying) {
   245→      memoryUsageWatcher.stopWatching(true)
   246→      memoryUsageWatcher.listener = null

   266→  override fun onApplyWindowInsets(insets: WindowInsetsCompat) {
   267→    super.onApplyWindowInsets(insets)

   269→    val imeInsets = insets.getInsets(WindowInsetsCompat.Type.ime())
   270→
   271→    _binding?.content?.bottomSheet?.setImeVisible(imeInsets.bottom > 0)
   272→    _binding?.contentCard?.updateLayoutParams<ViewGroup.LayoutParams> {
   273→      this.height = height - imeInsets.bottom

   275→
   276→    val isImeVisible = imeInsets.bottom > 0
   277→    if (this.isImeVisible != isImeVisible) {
   278→      this.isImeVisible = isImeVisible
   279→      onSoftInputChanged()
   280→    }

   308→    val packageName = onResult(this, intent) ?: return

   390→  private fun watchMemory() {

   394→  }
   395→
   396→  protected fun resetMemUsageChart() {
   397→    val processes = memoryUsageWatcher.getMemoryUsages()
   398→    val datasets = Array(processes.size) { index ->
   399→      LineDataSet(
   400→        List(MemoryUsageWatcher.MAX_USAGE_ENTRIES) { Entry(it.toFloat(), 0f) },
   401→        processes[index].pname
   402→      )
   403→    }
   404→
   405→    val bgColor = editorSurfaceContainerBackground
   406→    val textColor = resolveAttr(R.attr.colorOnSurface)
   407→
   408→    for ((index, proc) in processes.withIndex()) {
   409→      val dataset = datasets[index]
   410→      dataset.color = getMemUsageLineColorFor(proc)
   411→      dataset.setDrawIcons(false)
   412→      dataset.setDrawCircles(false)
   413→      dataset.setDrawCircleHole(false)
   414→      dataset.setDrawValues(false)
   415→      dataset.formLineWidth = 1f
   416→      dataset.formSize = 15f
   417→      dataset.isHighlightEnabled = false
   418→      pidToDatasetIdxMap[proc.pid] = index
   419→    }
   420→
   421→    binding.memUsageView.chart.setBackgroundColor(bgColor)
   422→
   423→    binding.memUsageView.chart.apply {
   424→      data = LineData(*datasets)
   425→      axisRight.textColor = textColor
   426→      axisLeft.textColor = textColor
   427→      legend.textColor = textColor
   428→
   429→      data.setValueTextColor(textColor)
   430→      setBackgroundColor(bgColor)
   431→      setGridBackgroundColor(bgColor)
   432→      notifyDataSetChanged()
   433→      invalidate()
   434→    }
   435→  }
   436→
   437→  private fun getMemUsageLineColorFor(proc: MemoryUsageWatcher.ProcessMemoryInfo): Int {
   438→    return when (proc.pname) {
   439→      PROC_IDE -> Color.BLUE
   440→      PROC_GRADLE_TOOLING -> Color.RED
   441→      PROC_GRADLE_DAEMON -> Color.GREEN
   442→      else -> throw IllegalArgumentException("Unknown process: $proc")
   443→    }
   444→  }

   609→  private fun handleUiDesignerResult(result: ActivityResult) {
   610→    if (result.resultCode != RESULT_OK || result.data == null) {
   611→      log.warn(
   612→        "UI Designer returned invalid result: resultCode={}, data={}", result.resultCode,
   613→        result.data
   614→      )
   615→      return
   616→    }
   617→    val generated = result.data!!.getStringExtra(UIDesignerActivity.RESULT_GENERATED_XML)
   618→    if (TextUtils.isEmpty(generated)) {
   619→      log.warn("UI Designer returned blank generated XML code")
   620→      return
   621→    }
   622→    val view = provideCurrentEditor()
   623→    val text = view?.editor?.text ?: run {
   624→      log.warn("No file opened to append UI designer result")
   625→      return
   626→    }
   627→    val endLine = text.lineCount - 1
   628→    text.replace(0, 0, endLine, text.getColumnCount(endLine), generated)
   629→  }

   718→  private fun appendClickableSpan(
   719→    sb: SpannableStringBuilder,
   720→    @StringRes textRes: Int,
   721→    span: ClickableSpan,
   722→  ) {
   723→    val str = getString(textRes)
   724→    val split = str.split("@@", limit = 3)
   725→    if (split.size != 3) {
   726→      // Not a valid format
   727→      sb.append(str)
   728→      sb.append('\n')
   729→      return
   730→    }
   731→    sb.append(split[0])
   732→    sb.append(split[1], span, Spanned.SPAN_EXCLUSIVE_EXCLUSIVE)
   733→    sb.append(split[2])
   734→    sb.append('\n')
   735→  }

   737→  private fun setupBottomSheet() {
   738→    editorBottomSheet = BottomSheetBehavior.from<View>(content.bottomSheet)
   739→    editorBottomSheet?.addBottomSheetCallback(object : BottomSheetCallback() {
   740→      override fun onStateChanged(bottomSheet: View, newState: Int) {
   741→        if (newState == BottomSheetBehavior.STATE_EXPANDED) {
   742→          val editor = provideCurrentEditor()
   743→          editor?.editor?.ensureWindowsDismissed()
   744→        }
   745→      }
   746→
   747→      override fun onSlide(bottomSheet: View, slideOffset: Float) {
   748→        content.apply {
   749→          val editorScale = 1 - slideOffset * (1 - EDITOR_CONTAINER_SCALE_FACTOR)
   750→          this.bottomSheet.onSlide(slideOffset)
   751→          this.viewContainer.scaleX = editorScale
   752→          this.viewContainer.scaleY = editorScale
   753→        }
   754→      }
   755→    })
   756→
   757→    val observer: OnGlobalLayoutListener = object : OnGlobalLayoutListener {
   758→      override fun onGlobalLayout() {
   759→        contentCardRealHeight = binding.contentCard.height
   760→        content.also {
   761→          it.realContainer.pivotX = it.realContainer.width.toFloat() / 2f
   762→          it.realContainer.pivotY =
   763→            (it.realContainer.height.toFloat() / 2f) + (systemBarInsets?.run { bottom - top }
   764→              ?: 0)
   765→          it.viewContainer.viewTreeObserver.removeOnGlobalLayoutListener(this)
   766→        }
   767→      }
   768→    }
   769→
   770→    content.apply {
   771→      viewContainer.viewTreeObserver.addOnGlobalLayoutListener(observer)
   772→      bottomSheet.setOffsetAnchor(editorAppBarLayout)
   773→    }
   774→  }
