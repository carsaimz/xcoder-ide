   155→   155→/**
   156→   156→ * A Compose command palette overlay for the XCoder editor.
   157→   157→ *
   158→   158→ * Features:
   159→   159→ * - Fuzzy search over all commands
   160→   160→ * - Category grouping with section headers
   161→   161→ * - Icon and shortcut display for each command
   162→   162→ * - Recent commands tracking
   163→   163→ * - Keyboard navigation (arrow keys + Enter)
   164→   164→ *
   165→   165→ * @param visible Whether the palette is currently shown.
   166→   166→ * @param onDismiss Callback when the palette is closed.
   167→   167→ * @param onCommand Callback when a command is executed, receiving the command ID.
   168→   168→ * @param extraCommands Additional commands to include beyond the built-in set.
   169→   169→ * @param modifier Modifier applied to the outer overlay.
   170→   170→ */
   171→   171→@OptIn(ExperimentalMaterial3Api::class)
   172→   172→@Composable
   173→   173→fun CommandPalette(
   174→   174→    visible: Boolean,
   175→   175→    onDismiss: () -> Unit,
   176→   176→    onCommand: (String) -> Unit,
   177→   177→    extraCommands: List<CommandItem> = emptyList(),
   178→   178→    modifier: Modifier = Modifier
   179→   179→) {
   180→   180→    var query by remember { mutableStateOf("") }
   181→   181→    var selectedIndex by remember { mutableIntStateOf(0) }
   182→   182→    var recentIds by remember { mutableStateOf(listOf<String>()) }
   183→   183→    val focusRequester = remember { FocusRequester() }
   184→   184→    val scope = rememberCoroutineScope()
   185→   185→
   186→   186→    // Combine built-in commands with any extra commands (extras take priority on duplicate IDs)
   187→   187→    val allCommands = remember(extraCommands) {
   188→   188→        val base = BuiltInCommands.all.toMutableList()
   189→   189→        for (extra in extraCommands) {
   190→   190→            val idx = base.indexOfFirst { it.id == extra.id }
   191→   191→            if (idx >= 0) base[idx] = extra else base.add(extra)
   192→   192→        }
   193→   193→        base
   194→   194→    }
   195→   195→
   196→   196→    // Filter commands by fuzzy query
   197→   197→    val filteredCommands = remember(query, allCommands) {
   198→   198→        if (query.isBlank()) {
   199→   199→            allCommands
   200→   200→        } else {
   201→   201→            allCommands.filter { cmd ->
   202→   202→                fuzzyMatch(query, cmd.label) ||
   203→   203→                    fuzzyMatch(query, cmd.id) ||
   204→   204→                    fuzzyMatch(query, cmd.category.displayName)
   205→   205→            }
   206→   206→        }
   207→   207→    }
   208→   208→
   209→   209→    // Group recent commands (those shown at the top when query is empty)
   210→   210→    val recentCommands = remember(recentIds, allCommands) {
   211→   211→        if (query.isNotBlank()) emptyList()
   212→   212→        else recentIds.mapNotNull { id -> allCommands.find { it.id == id } }
   213→   213→    }
   214→   214→
   215→   215→    // Show items: recent first (when no query), then filtered, grouped by category
   216→   216→    val displaySections = remember(filteredCommands, recentCommands, query) {
   217→   217→        val sections = mutableListOf<CommandPaletteSection>()
   218→   218→        if (recentCommands.isNotEmpty()) {
   219→   219→            sections.add(CommandPaletteSection("Recent", recentCommands))
   220→   220→        }
   221→   221→        if (query.isBlank() && recentCommands.isNotEmpty()) {
   222→   222→            val remaining = allCommands.filter { cmd ->
   223→   223→                recentIds.contains(cmd.id).not()
   224→   224→            }
   225→   225→            val grouped = remaining.groupBy { it.category }
   226→   226→            for (cat in CommandCategory.entries) {
   227→   227→                grouped[cat]?.let { cmds ->
   228→   228→                    if (cmds.isNotEmpty()) {
   229→   229→                        sections.add(CommandPaletteSection(cat.displayName, cmds))
   230→   230→                    }
   231→   231→                }
   232→   232→            }
   233→   233→        } else {
   234→   234→            val grouped = filteredCommands.groupBy { it.category }
   235→   235→            for (cat in CommandCategory.entries) {
   236→   236→                grouped[cat]?.let { cmds ->
   237→   237→                    if (cmds.isNotEmpty()) {
   238→   238→                        sections.add(CommandPaletteSection(cat.displayName, cmds))
   239→   239→                    }
   240→   240→                }
   241→   241→            }
   242→   242→        }
   243→   243→        sections
   244→   244→    }
   245→   245→
   246→   246→    // Flat list index helper for keyboard navigation
   247→   247→    val flatItemCount = remember(displaySections) {
   248→   248→        displaySections.sumOf { 1 + it.commands.size }
   249→   249→    }
   250→   250→
   251→   251→    // Ensure selected index is in bounds
   252→   252→    LaunchedEffect(flatItemCount) {
   253→   253→        if (selectedIndex >= flatItemCount) {
   254→   254→            selectedIndex = (flatItemCount - 1).coerceAtLeast(0)
   255→   255→        }
   256→   256→    }
   257→   257→
   258→   258→    // Auto-focus the search input when palette becomes visible
   259→   259→    LaunchedEffect(visible) {
   260→   260→        if (visible) {
   261→   261→            query = ""
   262→   262→            selectedIndex = 0
   263→   263→            delay(50)
   264→   264→            focusRequester.requestFocus()
   265→   265→        }
   266→   266→    }
   267→   267→
   268→   268→    AnimatedVisibility(
   269→   269→        visible = visible,
   270→   270→        enter = fadeIn(animationSpec = tween(100)) + slideInVertically(
   271→   271→            initialOffsetY = { -it / 4 },
   272→   272→            animationSpec = tween(150)
   273→   273→        ),
   274→   274→        exit = fadeOut(animationSpec = tween(80)) + slideOutVertically(
   275→   275→            targetOffsetY = { -it / 4 },
   276→   276→            animationSpec = tween(100)
   277→   277→        ),
   278→   278→        modifier = modifier
   279→   279→    ) {
   280→   280→        // Dimmed overlay backdrop
   281→   281→        Box(
   282→   282→            modifier = Modifier
   283→   283→                .fillMaxSize()
   284→   284→                .background(Color.Black.copy(alpha = 0.5f))
   285→   285→                .clickable(
   286→   286→                    indication = null,
   287→   287→                    interactionSource = remember { androidx.compose.foundation.interaction.MutableInteractionSource() }
   288→   288→                ) { onDismiss() },
   289→   289→            contentAlignment = Alignment.TopCenter
   290→   290→        ) {
   291→   291→            Card(
   292→   292→                modifier = Modifier
   293→   293→                    .padding(top = 80.dp)
   294→   294→                    .fillMaxWidth(0.92f)
   295→   295→                    .heightIn(max = 420.dp),
   296→   296→                shape = RoundedCornerShape(12.dp),
   297→   297→                colors = CardDefaults.cardColors(
   298→   298→                    containerColor = MaterialTheme.colorScheme.surfaceVariant
   299→   299→                ),
   300→   300→                elevation = CardDefaults.cardElevation(defaultElevation = 12.dp)
   301→   301→            ) {
   302→   302→                Column {
   303→   303→                    // Search input
   304→   304→                    OutlinedTextField(
   305→   305→                        value = query,
   306→   306→                        onValueChange = {
   307→   307→                            query = it
   308→   308→                            selectedIndex = 0
   309→   309→                        },
   310→   310→                        modifier = Modifier
   311→   311→                            .fillMaxWidth()
   312→   312→                            .focusRequester(focusRequester)
   313→   313→                            .focusable()
   314→   314→                            .onKeyEvent { event ->
   315→   315→                                handlePaletteKeyEvent(
   316→   316→                                    event = event,
   317→   317→                                    flatItemCount = flatItemCount,
   318→   318→                                    selectedIndex = selectedIndex,
   319→   319→                                    displaySections = displaySections,
   320→   320→                                    onSelectIndex = { selectedIndex = it },
   321→   321→                                    onExecute = { cmdId ->
   322→   322→                                        recentIds = (listOf(cmdId) + recentIds)
   323→   323→                                            .distinct()
   324→   324→                                            .take(10)
   325→   325→                                        onCommand(cmdId)
   326→   326→                                        onDismiss()
   327→   327→                                    },
   328→   328→                                    onDismiss = onDismiss
   329→   329→                                )
   330→   330→                            },
   331→   331→                        placeholder = {
   332→   332→                            Text(
   333→   333→                                text = if (query.startsWith(">"))
   334→   334→                                    "Type a command..."
   335→   335→                                else
   336→   336→                                    "Search commands by name, category, or shortcut...",
   337→   337→                                color = MaterialTheme.colorScheme.onSurfaceVariant.copy(alpha = 0.6f),
   338→   338→                                fontSize = 14.sp
   339→   339→                            )
   340→   340→                        },
   341→   341→                        leadingIcon = {
   342→   342→                            Icon(
   343→   343→                                imageVector = Icons.Default.Search,
   344→   344→                                contentDescription = null,
   345→   345→                                tint = MaterialTheme.colorScheme.onSurfaceVariant,
   346→   346→                                modifier = Modifier.size(18.dp)
   347→   347→                            )
   348→   348→                        },
   349→   349→                        singleLine = true,
   350→   350→                        textStyle = MaterialTheme.typography.bodyLarge.copy(
   351→   351→                            color = MaterialTheme.colorScheme.onSurface
   352→   352→                        ),
   353→   353→                        colors = OutlinedTextFieldDefaults.colors(
   354→   354→                            focusedBorderColor = MaterialTheme.colorScheme.primary,
   355→   355→                            unfocusedBorderColor = Color.Transparent,
   356→   356→                            cursorColor = MaterialTheme.colorScheme.primary
   357→   357→                        )
   358→   358→                    )
   359→   359→
   360→   360→                    HorizontalDivider(
   361→   361→                        color = MaterialTheme.colorScheme.outlineVariant.copy(alpha = 0.5f)
   362→   362→                    )
   363→   363→
   364→   364→                    // Command list
   365→   365→                    LazyColumn(
   366→   366→                        modifier = Modifier
   367→   367→                            .fillMaxWidth()
   368→   368→                            .weight(1f, fill = false),
   369→   369→                        contentPadding = PaddingValues(vertical = 4.dp)
   370→   370→                    ) {
   371→   371→                        var flatIndex = 0
   372→   372→                        displaySections.forEach { section ->
   373→   373→                            // Section header
   374→   374→                            item(key = "header_${section.title}") {
   375→   375→                                Text(
   376→   376→                                    text = section.title,
   377→   377→                                    style = MaterialTheme.typography.labelSmall.copy(
   378→   378→                                        fontWeight = FontWeight.SemiBold,
   379→   379→                                        letterSpacing = 1.sp
   380→   380→                                    ),
   381→   381→                                    color = MaterialTheme.colorScheme.onSurfaceVariant,
   382→   382→                                    modifier = Modifier.padding(horizontal = 16.dp, vertical = 6.dp)
   383→   383→                                )
   384→   384→                                flatIndex++ // header counts as an item for keyboard nav
   385→   385→                            }
   386→   386→
   387→   387→                            // Command items
   388→   388→                            itemsIndexed(section.commands) { index, cmd ->
   389→   389→                                val thisFlatIndex = flatIndex
   390→   390→                                flatIndex++
   391→   391→                                CommandItemRow(
   392→   392→                                    command = cmd,
   393→   393→                                    isSelected = thisFlatIndex == selectedIndex,
   394→   394→                                    onClick = {
   395→   395→                                        recentIds = (listOf(cmd.id) + recentIds)
   396→   396→                                            .distinct()
   397→   397→                                            .take(10)
   398→   398→                                        onCommand(cmd.id)
   399→   399→                                        onDismiss()
   400→   400→                                    },
   401→   401→                                    modifier = Modifier.clickable {
   402→   402→                                        recentIds = (listOf(cmd.id) + recentIds)
   403→   403→                                            .distinct()
   404→   404→                                            .take(10)
   405→   405→                                        onCommand(cmd.id)
   406→   406→                                        onDismiss()
   407→   407→                                    }
   408→   408→                                )
   409→   409→                            }
   410→   410→                        }
   411→   411→                    }
   412→   412→

   523→   523→/**
   524→   524→ * Fuzzy matching: returns true if all characters of [query] appear in [text]
   525→   525→ * in order (case-insensitive).
   526→   526→ */
   527→   527→internal fun fuzzyMatch(query: String, text: String): Boolean {
   528→   528→    val q = query.lowercase()
   529→   529→    val t = text.lowercase()
   530→   530→    var qi = 0
   531→   531→    for (ti in t.indices) {
   532→   532→        if (qi < q.length && t[ti] == q[qi]) {
   533→   533→            qi++
   534→   534→        }
   535→   535→    }
   536→   536→    return qi == q.length
   537→   537→}

   539→   539→/**
   540→   540→ * Handles keyboard events for the command palette.
   541→   541→ * Returns true if the event was consumed.
   542→   542→ */
   543→   543→private fun handlePaletteKeyEvent(
   544→   544→    event: KeyEvent,
   545→   545→    flatItemCount: Int,
   546→   546→    selectedIndex: Int,
   547→   547→    displaySections: List<CommandPaletteSection>,
   548→   548→    onSelectIndex: (Int) -> Unit,
   549→   549→    onExecute: (String) -> Unit,
   550→   550→    onDismiss: () -> Unit
   551→   551→): Boolean {
   552→   552→    when (event.type) {
   553→   553→        KeyEventType.KeyDown -> {
   554→   554→            when (event.key) {
   555→   555→                Key.DirectionDown -> {
   556→   556→                    val next = (selectedIndex + 1).coerceAtMost(flatItemCount - 1)
   557→   557→                    onSelectIndex(next)
   558→   558→                    return true
   559→   559→                }
   560→   560→                Key.DirectionUp -> {
   561→   561→                    val prev = (selectedIndex - 1).coerceAtLeast(0)
   562→   562→                    onSelectIndex(prev)
   563→   563→                    return true
   564→   564→                }
   565→   565→                Key.Enter -> {
   566→   566→                    val cmd = getCommandAtFlatIndex(displaySections, selectedIndex)
   567→   567→                    if (cmd != null) {
   568→   568→                        onExecute(cmd.id)
   569→   569→                    }
   570→   570→                    return true
   571→   571→                }
   572→   572→                Key.Escape -> {
   573→   573→                    onDismiss()
   574→   574→                    return true
   575→   575→                }
   576→   576→                else -> return false
   577→   577→            }
   578→   578→        }
   579→   579→        else -> return false
   580→   580→    }
   581→   581→}

   583→   583→/**
   584→   584→ * Retrieves the [CommandItem] at a given flat index across sections.
   585→   585→ * Section headers count as indices too.
   586→   586→ */
   587→   587→private fun getCommandAtFlatIndex(
   588→   588→    sections: List<CommandPaletteSection>,
   589→   589→    flatIndex: Int
   590→   590→): CommandItem? {
   591→   591→    var current = 0
   592→   592→    for (section in sections) {
   593→   593→        current++ // header
   594→   594→        if (flatIndex < current + section.commands.size) {
   595→   595→            return section.commands[flatIndex - current]
   596→   596→        }
   597→   597→        current += section.commands.size
   598→   598→    }
   599→   599→    return null
   600→   600→}
