  1010→    private String autoCreateOrReferenceString(String input) {
  1011→        if (input == null) return null;
  1012→        String value = input.trim();
  1013→        if (value.isEmpty()) return input;
  1014→        if (value.startsWith("@string/")) return value;
  1015→
  1016→        String stringsPath = wq.b(sc_id) + "/files/resource/values/strings.xml";
  1017→        StringsEditorManager sem = new StringsEditorManager();
  1018→        sem.sc_id = sc_id;
  1019→        ArrayList<HashMap<String, Object>> listMap = new ArrayList<>();
  1020→        String xml = FileUtil.readFileIfExist(stringsPath);
  1021→        sem.convertXmlStringsToListMap(xml, listMap);
  1022→
  1023→        for (HashMap<String, Object> map : listMap) {
  1024→            Object textObj = map.get("text");
  1025→            if (textObj != null && value.equals(String.valueOf(textObj))) {
  1026→                Object keyObj = map.get("key");
  1027→                if (keyObj != null) {
  1028→                    return "@string/" + String.valueOf(keyObj);
  1029→                }
  1030→            }
  1031→        }
  1032→
  1033→        String base = value.toLowerCase(Locale.ROOT)
  1034→                .replaceAll("[^a-z0-9]+", "_")
  1035→                .replaceAll("^_+|_+$", "");
  1036→        if (base.isEmpty()) base = "string_value";
  1037→        if (Character.isDigit(base.charAt(0))) base = "s_" + base;
  1038→
  1039→        String key = base;
  1040→        int suffix = 1;
  1041→        while (sem.isXmlStringsExist(listMap, key)) {
  1042→            key = base + "_" + suffix++;
  1043→        }
  1044→
  1045→        HashMap<String, Object> newMap = new HashMap<>();
  1046→        newMap.put("key", key);
  1047→        newMap.put("text", value);
  1048→        listMap.add(newMap);
  1049→
  1050→        XmlUtil.saveXml(stringsPath, sem.convertListMapToXmlStrings(listMap, sem.notesMap));
  1051→        return "@string/" + key;
  1052→    }
