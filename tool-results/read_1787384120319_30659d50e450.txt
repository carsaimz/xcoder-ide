    52→    data object DeleteLine : EditorAction()

    54→    data object MoveLineUp : EditorAction()

    56→    data object MoveLineDown : EditorAction()

   124→/**
   125→ * Enhanced language detection mapping file extensions to sora-editor Language objects.
   126→ *
   127→ * Uses the TextMate grammar scope names expected by sora-editor's [TextMateLanguage].
   128→ * Falls back to built-in languages (JavaLanguage) for Java where available,
   129→ * then to [EmptyLanguage] for unknown types.
   130→ *
   131→ * @param context Android context for loading TextMate grammars from assets.
   132→ * @param filePath File path used to determine the extension.
   133→ * @return A [Language] instance appropriate for the file type.
   134→ */
   135→fun detectLanguageForFile(context: Context, filePath: String): Language {
   136→    val ext = filePath.substringAfterLast('.', "").lowercase()
   137→    val scopeName = EXTENSION_TO_SCOPE[ext] ?: return EmptyLanguage()
   138→
   139→    // Use built-in Java language for best performance
   140→    if (ext == "java") {
   141→        return try { JavaLanguage() } catch (_: Exception) { createTextMate(scopeName) }
   142→    }
   143→
   144→    return createTextMate(scopeName) ?: EmptyLanguage()
   145→}
   146→
   147→/** Get the human-readable language name for a file extension. */
   148→fun getLanguageName(filePath: String): String {
   149→    val ext = filePath.substringAfterLast('.', "").lowercase()
   150→    return EXTENSION_TO_DISPLAY_NAME[ext] ?: ext.uppercase()
   151→}
   152→
   153→/** Get the comment style for a file (for toggle-comment). */
   154→fun getCommentStyle(filePath: String): CommentStyle {
   155→    val ext = filePath.substringAfterLast('.', "").lowercase()
   156→    return EXTENSION_TO_COMMENT[ext] ?: CommentStyle("// ")
   157→}

   159→private fun createTextMate(scopeName: String): Language? {
   160→    return try {
   161→        TextMateLanguage.create(scopeName, null)
   162→    } catch (_: Exception) {
   163→        null
   164→    }
   165→}

   449→/**
   450→ * Composable wrapper around sora-editor's [CodeEditor].
   451→ *
   452→ * This is the core editor component, analogous to AndroidIDE's [CodeEditorView]
   453→ * which wraps `IDEEditor` (extends `CodeEditor`) with event handling and
   454→ * preference binding.
   455→ *
   456→ * Key features following AndroidIDE patterns:
   457→ * - Language detection via [detectLanguageForFile] with TextMate grammars
   458→ * - Color scheme support (dark/light) via [SoraThemes]
   459→ * - Editor actions dispatched through [EditorAction] sealed class
   460→ * - Event callbacks for content, selection, and scroll changes
   461→ * - Configuration via immutable [EditorConfig]
   462→ * - Auto-indent, word wrap, line numbers, font size all configurable
   463→ *
   464→ * @param filePath Path of the file being edited (for language detection).
   465→ * @param contentText Initial text content.
   466→ * @param config Editor configuration (font size, tab size, color scheme, etc.).
   467→ * @param action Optional action to dispatch (one-shot, consumed after handling).
   468→ * @param callbacks Event callbacks for content/selection/scroll changes.
   469→ * @param editorRef Ref to expose the underlying [CodeEditor] for advanced use.
   470→ * @param modifier Compose modifier.
   471→ */
   472→@Composable
   473→fun SoraEditor(
   474→    filePath: String = "",
   475→    contentText: String = "",
   476→    config: EditorConfig = EditorConfig(),
   477→    action: EditorAction? = null,
   478→    callbacks: EditorEventCallbacks = EditorEventCallbacks(),
   479→    editorRef: androidx.compose.runtime.Ref<CodeEditor?>? = null,
   480→    modifier: Modifier = Modifier,
   481→) {
   482→    val context = androidx.compose.ui.platform.LocalContext.current
   483→
   484→    // Track whether we've already set initial content to avoid re-applying
   485→    // on recomposition. AndroidIDE handles this by checking if the file
   486→    // is the same as the one already loaded.
   487→    var lastSetFilePath by remember { mutableStateOf<String?>(null) }
   488→    var lastSetTextHash by remember { mutableIntStateOf(0) }
   489→
   490→    // Compute the current color scheme outside the factory/update
   491→    val currentScheme = remember(config.isDark) { SoraThemes.schemeFor(config.isDark) }
   492→
   493→    AndroidView(
   494→        factory = { ctx ->
   495→            CodeEditor(ctx).apply {
   496→                applyConfig(config, currentScheme)
   497→
   498→                // Set initial content
   499→                setText(contentText)
   500→                lastSetTextHash = contentText.hashCode()
   501→                lastSetFilePath = filePath
   502→
   503→                // Detect and set language
   504→                if (filePath.isNotEmpty()) {
   505→                    setLang(detectLanguageForFile(ctx, filePath))
   506→                }
   507→
   508→                // ── Event listeners (AndroidIDE pattern) ──────────────
   509→                // AndroidIDE registers listeners on CodeEditorView to track
   510→                // cursor position (for status bar), modifications (for tab
   511→                // indicator), and scroll (for minimap sync).
   512→
   513→                setOnTextChangedListener { _, _ ->
   514→                    callbacks.onContentChanged?.invoke(text.toString())
   515→                }
   516→
   517→                subscribeEvent(SelectionChangeEvent::class.java) { event, _ ->
   518→                    val cursor = this.cursor ?: return@subscribeEvent true
   519→                    val indexer = text.getIndexer()
   520→                    val line = indexer.getLineNumber(cursor.left) + 1
   521→                    val column = indexer.getColumnNumber(cursor.left) + 1
   522→                    val selStart = event.leftIndex
   523→                    val selEnd = event.rightIndex
   524→                    callbacks.onSelectionChanged?.invoke(line, column, selStart, selEnd)
   525→                    true
   526→                }
   527→
   528→                subscribeEvent(ScrollEvent::class.java) { event, _ ->
   529→                    callbacks.onScrollChanged?.invoke(event.dx, event.dy)
   530→                    true
   531→                }
   532→
   533→                // Expose via ref
   534→                editorRef?.value = this
   535→            }
   536→        },
   537→        update = { editor ->
   538→            // Update configuration properties that may have changed
   539→            editor.colorScheme = currentScheme
   540→            editor.textSize = config.fontSize
   541→            editor.setTabWidth(config.tabSize)
   542→            editor.isEditable = !config.readOnly
   543→            editor.isCursorVisible = !config.readOnly
   544→            editor.isWordWrap = config.wordWrap
   545→            editor.isLineNumberEnabled = config.showLineNumbers
   546→            editor.isMinimapEnabled = config.showMinimap
   547→            editor.isIndentGuideEnabled = config.showIndentGuides
   548→            editor.isStickyScrollEnabled = config.stickyScroll
   549→            editor.isHighlightCurrentLine = config.highlightCurrentLine
   550→            editor.isHighlightBracketPair = config.highlightBracketPair
   551→            editor.isAutoIndent = config.autoIndent
   552→            editor.isAutoCompletionEnabled = config.autoCompletion
   553→            editor.isSymbolCompletionEnabled = config.symbolCompletion
   554→            editor.isSmartBackspace = config.smartBackspace
   555→            editor.isPinchZoomEnabled = config.pinchZoom
   556→            config.fontTypeface?.let { editor.typefaceText = it }
   557→
   558→            // Handle content/switching (AndroidIDE pattern: only reset content
   559→            // when the file path changes, not on every recomposition)
   560→            if (filePath != lastSetFilePath) {
   561→                editor.setText(contentText)
   562→                lastSetTextHash = contentText.hashCode()
   563→                lastSetFilePath = filePath
   564→                if (filePath.isNotEmpty()) {
   565→                    editor.setLang(detectLanguageForFile(context, filePath))
   566→                }
   567→            }
   568→
   569→            // Handle dispatched action (one-shot)
   570→            action?.let { act ->
   571→                handleEditorAction(editor, act, filePath)
   572→                callbacks.onActionHandled?.invoke(act)
   573→            }
   574→
   575→            // Keep ref in sync
   576→            editorRef?.value = editor
   577→        },
   578→        modifier = modifier.fillMaxSize(),
   579→    )
   580→}

   582→// ── Internal helpers ────────────────────────────────────────────────────────
   583→
   584→/** Apply configuration to a fresh editor instance. */
   585→private fun CodeEditor.applyConfig(config: EditorConfig, scheme: EditorColorScheme) {
   586→    colorScheme = scheme
   587→    typefaceText = config.fontTypeface ?: Typeface.MONOSPACE
   588→    textSize = config.fontSize
   589→    setTabWidth(config.tabSize)
   590→    isWordWrap = config.wordWrap
   591→    isCursorVisible = !config.readOnly
   592→    isEditable = !config.readOnly
   593→    isHighlightCurrentLine = config.highlightCurrentLine
   594→    isHighlightBracketPair = config.highlightBracketPair
   595→    isHighlightMatchingDelimiters = true
   596→    isAutoCompletionEnabled = config.autoCompletion
   597→    isAutoIndent = config.autoIndent
   598→    isSmartBackspace = config.smartBackspace
   599→    isShowLineNumber = config.showLineNumbers
   600→    isLineNumberEnabled = config.showLineNumbers
   601→    isPinchZoomEnabled = config.pinchZoom
   602→    isStickyScrollEnabled = config.stickyScroll
   603→    isIndentGuideEnabled = config.showIndentGuides
   604→    isSymbolCompletionEnabled = config.symbolCompletion
   605→    isMinimapEnabled = config.showMinimap
   606→}
   607→
   608→/**
   609→ * Dispatch an [EditorAction] to a [CodeEditor].
   610→ * This is the central action handler, following AndroidIDE's command dispatch pattern.
   611→ */
   612→fun handleEditorAction(editor: CodeEditor, action: EditorAction, filePath: String) {
   613→    when (action) {
   614→        is EditorAction.Undo -> editor.undo()
   615→        is EditorAction.Redo -> editor.redo()
   616→        is EditorAction.Cut -> editor.copyText()  // sora-editor handles cut
   617→        is EditorAction.Copy -> editor.copyText()
   618→        is EditorAction.Paste -> { /* handled by system clipboard */ }
   619→        is EditorAction.SelectAll -> editor.selectAll()
   620→        is EditorAction.FormatCode -> editor.formatCode()
   621→
   622→        is EditorAction.ToggleComment -> toggleComment(editor, filePath)
   623→        is EditorAction.DuplicateLine -> duplicateLine(editor)
   624→        is EditorAction.DeleteLine -> deleteLine(editor)
   625→        is EditorAction.MoveLineUp -> moveLineUp(editor)
   626→        is EditorAction.MoveLineDown -> moveLineDown(editor)
   627→
   628→        is EditorAction.ToggleWordWrap -> { editor.isWordWrap = !editor.isWordWrap }
   629→        is EditorAction.ToggleLineNumbers -> {
   630→            editor.isLineNumberEnabled = !editor.isLineNumberEnabled
   631→        }
   632→        is EditorAction.ToggleMinimap -> {
   633→            editor.isMinimapEnabled = !editor.isMinimapEnabled
   634→        }
   635→        is EditorAction.ToggleIndentGuides -> {
   636→            editor.isIndentGuideEnabled = !editor.isIndentGuideEnabled
   637→        }
   638→        is EditorAction.ToggleStickyScroll -> {
   639→            editor.isStickyScrollEnabled = !editor.isStickyScrollEnabled
   640→        }
   641→
   642→        is EditorAction.FontSizeIncrease -> {
   643→            editor.textSize = editor.textSize + 1f
   644→        }
   645→        is EditorAction.FontSizeDecrease -> {
   646→            editor.textSize = (editor.textSize - 1f).coerceAtLeast(8f)
   647→        }
   648→        is EditorAction.FontSizeReset -> {
   649→            editor.textSize = action.defaultSize
   650→        }
   651→
   652→        is EditorAction.Search -> {
   653→            editor.searcher.search(action.query, action.caseSensitive, action.regex)
   654→        }
   655→        is EditorAction.Replace -> {
   656→            editor.searcher.replaceCurrent(action.query, action.replacement)
   657→        }
   658→        is EditorAction.ReplaceAll -> {
   659→            editor.searcher.replaceAll(action.query, action.replacement)
   660→        }
   661→        is EditorAction.SearchNext -> {
   662→            if (action.forward) editor.searcher.gotoNext() else editor.searcher.gotoPrevious()
   663→        }
   664→
   665→        is EditorAction.GoToLine -> goToLine(editor, action.line)
   666→        is EditorAction.InsertText -> insertAtCursor(editor, action.text)
   667→        is EditorAction.SetText -> editor.setText(action.text)
   668→    }
   669→}

   671→// ── Line/cursor operations ──────────────────────────────────────────────────
   672→
   673→/** Get the 0-indexed line number at the cursor. */
   674→private fun cursorLineIndex(editor: CodeEditor): Int {
   675→    val cursor = editor.cursor ?: return 0
   676→    return editor.text.getIndexer().getLineNumber(cursor.left)
   677→}
   678→
   679→/** Get the character index at the end of the cursor's line. */
   680→private fun cursorLineEnd(editor: CodeEditor): Int {
   681→    return editor.text.getLineEnd(cursorLineIndex(editor))
   682→}
   683→
   684→/** Insert text at the current cursor position. */
   685→private fun insertAtCursor(editor: CodeEditor, text: String) {
   686→    editor.cursor?.let { cursor ->
   687→        editor.text.insert(cursor.left, text)
   688→    }
   689→}
   690→
   691→/** Go to a specific line (1-indexed) and place cursor at column 1. */
   692→private fun goToLine(editor: CodeEditor, line: Int) {
   693→    val clampedLine = line.coerceIn(1, editor.text.lineCount)
   694→    val position = editor.text.getIndexer().getCharPosition(clampedLine - 1, 0)
   695→    editor.cursor?.setLeft(position)
   696→    editor.setSelection(position, position)
   697→}
   698→
   699→/** Duplicate the current line below. */
   700→private fun duplicateLine(editor: CodeEditor) {
   701→    val text = editor.text
   702→    val lineIndex = cursorLineIndex(editor)
   703→    val lineContent = text.getLine(lineIndex)
   704→    val endPos = text.getLineEnd(lineIndex)
   705→    // Insert after the current line's newline
   706→    val insertPos = if (endPos < text.length && text[endPos] == '\n') {
   707→        endPos + 1
   708→    } else {
   709→        endPos
   710→    }
   711→    text.insert(insertPos, "$lineContent\n")
   712→}
   713→
   714→/** Delete the current line entirely. */
   715→private fun deleteLine(editor: CodeEditor) {
   716→    val text = editor.text
   717→    val lineIndex = cursorLineIndex(editor)
   718→    val start = text.getLineStart(lineIndex)
   719→    var end = text.getLineEnd(lineIndex)
   720→    // Also consume the trailing newline if present
   721→    if (end < text.length && text[end] == '\n') end++
   722→    text.delete(start, end)
   723→}
   724→
   725→/** Swap the current line with the line above. */
   726→private fun moveLineUp(editor: CodeEditor) {
   727→    val text = editor.text
   728→    val lineIndex = cursorLineIndex(editor)
   729→    if (lineIndex <= 0) return
   730→    val curLine = text.getLine(lineIndex)
   731→    val prevLine = text.getLine(lineIndex - 1)
   732→    val prevStart = text.getLineStart(lineIndex - 1)
   733→    var curEnd = text.getLineEnd(lineIndex)
   734→    if (curEnd < text.length && text[curEnd] == '\n') curEnd++
   735→    text.replace(prevStart, curEnd.coerceAtMost(text.length), "$curLine\n$prevLine")
   736→}
   737→
   738→/** Swap the current line with the line below. */
   739→private fun moveLineDown(editor: CodeEditor) {
   740→    val text = editor.text
   741→    val lineIndex = cursorLineIndex(editor)
   742→    if (lineIndex >= text.lineCount - 1) return
   743→    val curLine = text.getLine(lineIndex)
   744→    val nextLine = text.getLine(lineIndex + 1)
   745→    val curStart = text.getLineStart(lineIndex)
   746→    var nextEnd = text.getLineEnd(lineIndex + 1)
   747→    if (nextEnd < text.length && text[nextEnd] == '\n') nextEnd++
   748→    text.replace(curStart, nextEnd.coerceAtMost(text.length), "$nextLine\n$curLine")
   749→}

   751→/**
   752→ * Toggle line comment on the current line.
   753→ * Uses language-aware comment detection following AndroidIDE's pattern.
   754→ */
   755→private fun toggleComment(editor: CodeEditor, filePath: String) {
   756→    val text = editor.text
   757→    val lineIndex = cursorLineIndex(editor)
   758→    val style = getCommentStyle(filePath)
   759→    val line = text.getLine(lineIndex)
   760→    val lineStart = text.getLineStart(lineIndex)
   761→
   762→    // For line comments, toggle the prefix
   763→    val prefix = style.linePrefix.trimEnd()
   764→    if (line.trimStart().startsWith(prefix)) {
   765→        // Uncomment: find and remove the prefix
   766→        val trimmedStart = line.indexOf(prefix)
   767→        if (trimmedStart >= 0) {
   768→            text.delete(lineStart + trimmedStart, lineStart + trimmedStart + prefix.length)
   769→        }
   770→    } else {
   771→        // Comment: insert prefix at the start of line content (preserving indent)
   772→        val indent = line.takeWhile { it == ' ' || it == '\t' }
   773→        val insertPos = lineStart + indent.length
   774→        text.insert(insertPos, style.linePrefix)
   775→    }
   776→}

   780→/** Get the current cursor position as (line, column), both 1-indexed. */
   781→fun getCursorPosition(editor: CodeEditor): Pair<Int, Int> {
   782→    val cursor = editor.cursor ?: return 1 to 1
   783→    val indexer = editor.text.getIndexer()
   784→    val line = indexer.getLineNumber(cursor.left) + 1
   785→    val column = indexer.getColumnNumber(cursor.left) + 1
   786→    return line to column
   787→}

   789→/** Get the total line count. */
   790→fun getLineCount(editor: CodeEditor): Int = editor.text.lineCount

   792→/** Get the current selection range, or null if no selection. */
   793→fun getSelectionRange(editor: CodeEditor): Pair<Int, Int>? {
   794→    val cursor = editor.cursor ?: return null
   795→    val left = cursor.left
   796→    val right = cursor.right
   797→    return if (left != right) minOf(left, right) to maxOf(left, right) else null
   798→}
