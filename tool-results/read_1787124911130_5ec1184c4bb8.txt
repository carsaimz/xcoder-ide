   206→        // ─── Utility: detect mode from filename ───
   207→        function detectLanguage(filename) {
   208→            if (!filename) return 'text';
   209→            var lower = filename.toLowerCase();
   210→
   211→            // Check for compound extensions like .gradle.kts
   212→            if (lower.endsWith('.gradle.kts')) return 'kotlin';
   213→            if (lower.endsWith('.build.gradle.kts')) return 'kotlin';
   214→
   215→            // Check direct language name
   216→            if (languageModes[lower]) return lower;
   217→
   218→            // Extract extension
   219→            var parts = lower.split('.');
   220→            if (parts.length > 1) {
   221→                var ext = parts[parts.length - 1];
   222→                if (extensionToMode[ext]) return extensionToMode[ext];
   223→            }
   224→
   225→            return 'text';
   226→        }

   228→        function modePath(language) {
   229→            if (languageModes[language]) return languageModes[language];
   230→            return 'ace/mode/text';
   231→        }

   468→        // ─── Search panel ───
   469→        function setupSearchPanel() {
   470→            var searchInput = document.getElementById('search-input');
   471→            var replaceInput = document.getElementById('replace-input');
   472→            var searchCount = document.getElementById('search-count');
   473→            var btnPrev = document.getElementById('btn-prev');
   474→            var btnNext = document.getElementById('btn-next');
   475→            var btnReplace = document.getElementById('btn-replace');
   476→            var btnReplaceAll = document.getElementById('btn-replace-all');
   477→            var btnToggleReplace = document.getElementById('btn-toggle-replace');
   478→            var btnCloseSearch = document.getElementById('btn-close-search');
   479→            var panel = document.getElementById('search-panel');
   480→            var replaceVisible = false;
   481→
   482→            function updateSearchCount() {
   483→                var query = searchInput.value;
   484→                if (!query) {
   485→                    searchCount.textContent = '';
   486→                    return;
   487→                }
   488→                var count = 0;
   489→                var content = editor.getValue();
   490→                var pos = content.indexOf(query);
   491→                while (pos !== -1) {
   492→                    count++;
   493→                    pos = content.indexOf(query, pos + 1);
   494→                }
   495→                if (count === 0) {
   496→                    searchCount.textContent = 'No results';
   497→                } else {
   498→                    searchCount.textContent = count + ' found';
   499→                }
   500→            }
   501→
   502→            searchInput.addEventListener('input', function() {
   503→                var query = searchInput.value;
   504→                editor.find(query, {
   505→                    backwards: false,
   506→                    wrap: true,
   507→                    caseSensitive: false,
   508→                    wholeWord: false,
   509→                    regExp: false,
   510→                    showCursor: true
   511→                });
   512→                updateSearchCount();
   513→            });
   514→
   515→            searchInput.addEventListener('keydown', function(e) {
   516→                if (e.key === 'Enter') {
   517→                    if (e.shiftKey) {
   518→                        editor.findPrevious();
   519→                    } else {
   520→                        editor.findNext();
   521→                    }
   522→                } else if (e.key === 'Escape') {
   523→                    panel.classList.remove('visible');
   524→                    editor.focus();
   525→                }
   526→            });
   527→
   528→            btnPrev.addEventListener('click', function() {
   529→                editor.findPrevious();
   530→            });
   531→
   532→            btnNext.addEventListener('click', function() {
   533→                editor.findNext();
   534→            });
   535→
   536→            btnToggleReplace.addEventListener('click', function() {
   537→                replaceVisible = !replaceVisible;
   538→                replaceInput.style.display = replaceVisible ? 'inline-block' : 'none';
   539→                btnReplace.style.display = replaceVisible ? 'inline-block' : 'none';
   540→                btnReplaceAll.style.display = replaceVisible ? 'inline-block' : 'none';
   541→                if (replaceVisible) {
   542→                    replaceInput.focus();
   543→                }
   544→            });
   545→
   546→            btnReplace.addEventListener('click', function() {
   547→                var replacement = replaceInput.value;
   548→                editor.replace(replacement);
   549→                updateSearchCount();
   550→            });
   551→
   552→            btnReplaceAll.addEventListener('click', function() {
   553→                var query = searchInput.value;
   554→                var replacement = replaceInput.value;
   555→                if (query) {
   556→                    editor.findAll(query, {
   557→                        wrap: true,
   558→                        caseSensitive: false,
   559→                        wholeWord: false,
   560→                        regExp: false
   561→                    });
   562→                    editor.replace(replacement);
   563→                    updateSearchCount();
   564→                }
   565→            });
   566→
   567→            btnCloseSearch.addEventListener('click', function() {
   568→                panel.classList.remove('visible');
   569→                editor.focus();
   570→            });
   571→        }

   573→        function showSearchPanel(withReplace) {
   574→            var panel = document.getElementById('search-panel');
   575→            panel.classList.add('visible');

   577→            if (withReplace) {
   578→                var btnToggle = document.getElementById('btn-toggle-replace');
   579→                if (!document.getElementById('replace-input').style.display ||
   580→                    document.getElementById('replace-input').style.display === 'none') {
   581→                    btnToggle.click();
   582→                }
   583→            }
   584→
   585→            var selectedText = editor.getSelectedText();
   586→            if (selectedText && selectedText.length < 100) {
   587→                var searchInput = document.getElementById('search-input');
   588→                searchInput.value = selectedText;
   589→                // Trigger search
   590→                editor.find(selectedText, {
   591→                    backwards: false,
   592→                    wrap: true,
   593→                    caseSensitive: false,
   594→                    wholeWord: false,
   595→                    regExp: false,
   596→                    showCursor: true
   597→                });
   598→            }
   599→
   600→            document.getElementById('search-input').focus();
   601→            document.getElementById('search-input').select();
   602→        }

   611→                }
   612→            }

   624→        window.loadFile = function(content, language, readonly) {
   625→            if (!editor) return;
   626→            currentLanguage = language || 'text';
   627→            isReadOnly = readonly === true || readonly === 'true';
   628→
   629→            editor.setReadOnly(isReadOnly);
   630→            editor.setValue(content || '', -1);
   631→            editor.clearSelection();
   632→            editor.session.setMode(modePath(currentLanguage));
   633→            editor.gotoLine(1);
   634→            editor.focus();
   635→
   636→            notifyKotlin('onConsoleLog', 'File loaded: language=' + currentLanguage + ', readonly=' + isReadOnly);
   637→        };

   723→        window.search = function(query, optionsJson) {
   724→            if (!editor || !query) return '0';
   725→            var options = {};
   726→            if (optionsJson) {
   727→                try {
   728→                    options = JSON.parse(optionsJson);
   729→                } catch (e) {
   730→                    options = {};
   731→                }
   732→            }
   733→            editor.search(query, {
   734→                backwards: options.backwards || false,
   735→                wrap: options.wrap !== false,
   736→                caseSensitive: options.caseSensitive || false,
   737→                wholeWord: options.wholeWord || false,
   738→                regExp: options.regExp || false,
   739→                showCursor: true
   740→            });
   741→
   742→            // Count results
   743→            var count = 0;
   744→            var content = editor.getValue();
   745→            var caseSensitive = options.caseSensitive || false;
   746→            var searchContent = caseSensitive ? content : content.toLowerCase();
   747→            var searchQuery = caseSensitive ? query : query.toLowerCase();
   748→            var pos = searchContent.indexOf(searchQuery);
   749→            while (pos !== -1) {
   750→                count++;
   751→                pos = searchContent.indexOf(searchQuery, pos + 1);
   752→            }
   753→            return String(count);
   754→        };

   761→            while (pos !== -1) {
   762→                count++;
   763→                pos = content.indexOf(query, pos + 1);
   764→            }
   765→            var newValue = content.split(query).join(replacement);
   766→            editor.setValue(newValue, -1);
   767→            editor.clearSelection();
   768→            notifyKotlin('onContentChanged', newValue);
   769→            return String(count);
   770→        };

   787→            var formatted = [];
   788→            for (var i = 0; i < lines.length; i++) {
   789→                formatted.push(lines[i].replace(/\s+$/, ''));
   790→            }
   791→            var result = formatted.join('\n');
   792→            if (result.length > 0 && !result.endsWith('\n')) {
   793→                result += '\n';
   794→            }
   795→            editor.setValue(result, -1);
   796→            editor.clearSelection();
   797→            editor.gotoLine(1);
