   207→   207→        function detectLanguage(filename) {
   208→   208→            if (!filename) return 'text';
   209→   209→            var lower = filename.toLowerCase();
   210→   210→
   211→   211→            // Check for compound extensions like .gradle.kts
   212→   212→            if (lower.endsWith('.gradle.kts')) return 'kotlin';
   213→   213→            if (lower.endsWith('.build.gradle.kts')) return 'kotlin';
   214→   214→
   215→   215→            // Check direct language name
   216→   216→            if (languageModes[lower]) return lower;
   217→   217→
   218→   218→            // Extract extension
   219→   219→            var parts = lower.split('.');
   220→   220→            if (parts.length > 1) {
   221→   221→                var ext = parts[parts.length - 1];
   222→   222→                if (extensionToMode[ext]) return extensionToMode[ext];
   223→   223→            }
   224→   224→
   225→   225→            return 'text';
   226→   226→        }

   482→   482→            function updateSearchCount() {
   483→   483→                var query = searchInput.value;
   484→   484→                if (!query) {
   485→   485→                    searchCount.textContent = '';
   486→   486→                    return;
   487→   487→                }
   488→   488→                var count = 0;
   489→   489→                var content = editor.getValue();
   490→   490→                var pos = content.indexOf(query);
   491→   491→                while (pos !== -1) {
   492→   492→                    count++;
   493→   493→                    pos = content.indexOf(query, pos + 1);
   494→   494→                }
   495→   495→                if (count === 0) {
   496→   496→                    searchCount.textContent = 'No results';
   497→   497→                } else {
   498→   498→                    searchCount.textContent = count + ' found';
   499→   499→                }
   500→   500→            }
   501→   501→
   502→   502→            searchInput.addEventListener('input', function() {
   503→   503→                var query = searchInput.value;
   504→   504→                editor.find(query, {
   505→   505→                    backwards: false,
   506→   506→                    wrap: true,
   507→   507→                    caseSensitive: false,
   508→   508→                    wholeWord: false,
   509→   509→                    regExp: false,
   510→   510→                    showCursor: true
   511→   511→                });
   512→   512→                updateSearchCount();
   513→   513→            });
   514→   514→
   515→   515→            searchInput.addEventListener('keydown', function(e) {
   516→   516→                if (e.key === 'Enter') {
   517→   517→                    if (e.shiftKey) {
   518→   518→                        editor.findPrevious();
   519→   519→                    } else {
   520→   520→                        editor.findNext();

   725→   725→            var options = {};
   726→   726→            if (optionsJson) {
   727→   727→                try {

   729→   729→                } catch (e) {
   730→   730→                    options = {};
   731→   731→                }
   732→   732→            }
   733→   733→            editor.search(query, {
   734→   734→                backwards: options.backwards || false,
   735→   735→                wrap: options.wrap !== false,
   736→   736→                caseSensitive: options.caseSensitive || false,
   737→   737→                wholeWord: options.wholeWord || false,
   738→   738→                regExp: options.regExp || false,
   739→   739→                showCursor: true
   740→   740→            });
   741→   741→
   742→   742→            // Count results
   743→   743→            var count = 0;
   744→   744→            var content = editor.getValue();
   745→   745→            var caseSensitive = options.caseSensitive || false;
   746→   746→            var searchContent = caseSensitive ? content : content.toLowerCase();
   747→   747→            var searchQuery = caseSensitive ? query : query.toLowerCase();
   748→   748→            var pos = searchContent.indexOf(searchQuery);
   749→   749→            while (pos !== -1) {
   750→   750→                count++;
   751→   751→                pos = searchContent.indexOf(searchQuery, pos + 1);
   752→   752→            }
   753→   753→            return String(count);
   754→   754→        };
