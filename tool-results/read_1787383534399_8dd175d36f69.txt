   142→            @Override
   143→            public boolean onUp(MotionEvent event) {
   144→                mScrollRemainder = 0.0f;
   145→                if (mEmulator != null && mEmulator.isMouseTrackingActive() && !event.isFromSource(InputDevice.SOURCE_MOUSE) && !isSelectingText() && !scrolledWithFinger) {
   146→                    // Quick event processing when mouse tracking is active - do not wait for check of double tapping
   147→                    // for zooming.
   148→                    sendMouseEventCode(event, TerminalEmulator.MOUSE_LEFT_BUTTON, true);
   149→                    sendMouseEventCode(event, TerminalEmulator.MOUSE_LEFT_BUTTON, false);
   150→                    return true;
   151→                }
   152→                scrolledWithFinger = false;
   153→                return false;
   154→            }

   157→            public boolean onSingleTapUp(MotionEvent event) {
   158→                if (mEmulator == null) return true;
   159→
   160→                if (isSelectingText()) {
   161→                    stopTextSelectionMode();

   169→            @Override
   170→            public boolean onScroll(MotionEvent e, float distanceX, float distanceY) {
   171→                if (mEmulator == null) return true;
   172→                if (mEmulator.isMouseTrackingActive() && e.isFromSource(InputDevice.SOURCE_MOUSE)) {
   173→                    // If moving with mouse pointer while pressing button, report that instead of scroll.
   174→                    // This means that we never report moving with button press-events for touch input,
   175→                    // since we cannot just start sending these events without a starting press event,
   176→                    // which we do not do for touch input, only mouse in onTouchEvent().
   177→                    sendMouseEventCode(e, TerminalEmulator.MOUSE_LEFT_BUTTON_MOVED, true);
   178→                } else {
   179→                    scrolledWithFinger = true;
   180→                    distanceY += mScrollRemainder;
   181→                    int deltaRows = (int) (distanceY / mRenderer.mFontLineSpacing);
   182→                    mScrollRemainder = distanceY - deltaRows * mRenderer.mFontLineSpacing;
   183→                    doScroll(e, deltaRows);
   184→                }
   185→                return true;
   186→            }

   189→            public boolean onScale(float focusX, float focusY, float scale) {
   190→                if (mEmulator == null || isSelectingText()) return true;
   191→                mScaleFactor *= scale;
   192→                mScaleFactor = mClient.onScale(mScaleFactor);

   196→            @Override
   197→            public boolean onFling(final MotionEvent e2, float velocityX, float velocityY) {
   198→                if (mEmulator == null) return true;
   199→                // Do not start scrolling until last fling has been taken care of:
   200→                if (!mScroller.isFinished()) return true;
   201→
   202→                final boolean mouseTrackingAtStartOfFling = mEmulator.isMouseTrackingActive();
   203→                float SCALE = 0.25f;
   204→                if (mouseTrackingAtStartOfFling) {
   205→                    mScroller.fling(0, 0, 0, -(int) (velocityY * SCALE), 0, 0, -mEmulator.mRows / 2, mEmulator.mRows / 2);
   206→                } else {
   207→                    mScroller.fling(0, mTopRow, 0, -(int) (velocityY * SCALE), 0, 0, -mEmulator.getScreen().getActiveTranscriptRows(), 0);
   208→                }
   209→
   210→                post(new Runnable() {
   211→                    private int mLastY = 0;
   212→
   213→                    @Override
   214→                    public void run() {
   215→                        if (mouseTrackingAtStartOfFling != mEmulator.isMouseTrackingActive()) {
   216→                            mScroller.abortAnimation();
   217→                            return;
   218→                        }
   219→                        if (mScroller.isFinished()) return;
   220→                        boolean more = mScroller.computeScrollOffset();
   221→                        int newY = mScroller.getCurrY();
   222→                        int diff = mouseTrackingAtStartOfFling ? (newY - mLastY) : (newY - mTopRow);
   223→                        doScroll(e2, diff);
   224→                        mLastY = newY;
   225→                        if (more) post(this);
   226→                    }
   227→                });
   228→
   229→                return true;
   230→            }

   401→    @SuppressLint("ClickableViewAccessibility")
   402→    @Override
   403→    @TargetApi(23)
   404→    public boolean onTouchEvent(MotionEvent event) {
   405→        if (mEmulator == null) return true;
   406→        final int action = event.getAction();
   407→
   408→        if (isSelectingText()) {
   409→            updateFloatingToolbarVisibility(event);
   410→            mGestureRecognizer.onTouchEvent(event);
   411→            return true;
   412→        } else if (event.isFromSource(InputDevice.SOURCE_MOUSE)) {
   413→            if (event.isButtonPressed(MotionEvent.BUTTON_SECONDARY)) {
   414→                if (action == MotionEvent.ACTION_DOWN) showContextMenu();
   415→                return true;
   416→            } else if (event.isButtonPressed(MotionEvent.BUTTON_TERTIARY)) {
   417→                ClipboardManager clipboardManager = (ClipboardManager) getContext().getSystemService(Context.CLIPBOARD_SERVICE);
   418→                ClipData clipData = clipboardManager.getPrimaryClip();
   419→                if (clipData != null) {
   420→                    ClipData.Item clipItem = clipData.getItemAt(0);
   421→                    if (clipItem != null) {
   422→                        CharSequence text = clipItem.coerceToText(getContext());
   423→                        if (!TextUtils.isEmpty(text)) mEmulator.paste(text.toString());
   424→                    }
   425→                }
   426→            } else if (mEmulator.isMouseTrackingActive()) { // BUTTON_PRIMARY.
   427→                switch (event.getAction()) {
   428→                    case MotionEvent.ACTION_DOWN:
   429→                    case MotionEvent.ACTION_UP:
   430→                        sendMouseEventCode(event, TerminalEmulator.MOUSE_LEFT_BUTTON, event.getAction() == MotionEvent.ACTION_DOWN);
   431→                        break;
   432→                    case MotionEvent.ACTION_MOVE:
   433→                        sendMouseEventCode(event, TerminalEmulator.MOUSE_LEFT_BUTTON_MOVED, true);
   434→                        break;
   435→                }
   436→            }
   437→        }
   438→
   439→        mGestureRecognizer.onTouchEvent(event);
   440→        return true;
   441→    }

   443→    @Override
   444→    public boolean onKeyPreIme(int keyCode, KeyEvent event) {
   445→        if (TERMINAL_VIEW_KEY_LOGGING_ENABLED)
   446→            mClient.logInfo(LOG_TAG, "onKeyPreIme(keyCode=" + keyCode + ", event=" + event + ")");
   447→        if (keyCode == KeyEvent.KEYCODE_BACK) {
   448→            cancelRequestAutoFill();
   449→            if (isSelectingText()) {
   450→                stopTextSelectionMode();
   451→                return true;
   452→            } else if (mClient.shouldBackButtonBeMappedToEscape()) {
   453→                // Intercept back button to treat it as escape:
   454→                switch (event.getAction()) {
   455→                    case KeyEvent.ACTION_DOWN:
   456→                        return onKeyDown(keyCode, event);
   457→                    case KeyEvent.ACTION_UP:
   458→                        return onKeyUp(keyCode, event);
   459→                }
   460→            }
   461→        } else if (mClient.shouldUseCtrlSpaceWorkaround() &&
   462→                   keyCode == KeyEvent.KEYCODE_SPACE && event.isCtrlPressed()) {
   463→            /* ctrl+space does not work on some ROMs without this workaround.
   464→               However, this breaks it on devices where it works out of the box. */
   465→            return onKeyDown(keyCode, event);
   466→        }
   467→        return super.onKeyPreIme(keyCode, event);
   468→    }

   470→    /**
   471→     * Key presses in software keyboards will generally NOT trigger this listener, although some
   472→     * may elect to do so in some situations. Do not rely on this to catch software key presses.
   473→     * Gboard calls this when shouldEnforceCharBasedInput() is disabled (InputType.TYPE_NULL) instead
   474→     * of calling commitText(), with deviceId=-1. However, Hacker's Keyboard, OpenBoard, LG Keyboard
   475→     * call commitText().
   476→     *
   477→     * This function may also be called directly without android calling it, like by
   478→     * `TerminalExtraKeys` which generates a KeyEvent manually which uses {@link KeyCharacterMap#VIRTUAL_KEYBOARD}
   479→     * as the device (deviceId=-1), as does Gboard. That would normally use mappings defined in
   480→     * `/system/usr/keychars/Virtual.kcm`. You can run `dumpsys input` to find the `KeyCharacterMapFile`
   481→     * used by virtual keyboard or hardware keyboard. Note that virtual keyboard device is not the
   482→     * same as software keyboard, like Gboard, etc. Its a fake device used for generating events and
   483→     * for testing.
   484→     *
   485→     * We handle shift key in `commitText()` to convert codepoint to uppercase case there with a
   486→     * call to {@link Character#toUpperCase(int)}, but here we instead rely on getUnicodeChar() for
   487→     * conversion of keyCode, for both hardware keyboard shift key (via effectiveMetaState) and
   488→     * `mClient.readShiftKey()`, based on value in kcm files.
   489→     * This may result in different behaviour depending on keyboard and android kcm files set for the
   490→     * InputDevice for the event passed to this function. This will likely be an issue for non-english
   491→     * languages since `Virtual.kcm` in english only by default or at least in AOSP. For both hardware
   492→     * shift key (via effectiveMetaState) and `mClient.readShiftKey()`, `getUnicodeChar()` is used
   493→     * for shift specific behaviour which usually is to uppercase.
   494→     *
   495→     * For fn key on hardware keyboard, android checks kcm files for hardware keyboards, which is
   496→     * `Generic.kcm` by default, unless a vendor specific one is defined. The event passed will have
   497→     * {@link KeyEvent#META_FUNCTION_ON} set. If the kcm file only defines a single character or unicode
   498→     * code point `\\uxxxx`, then only one event is passed with that value. However, if kcm defines
   499→     * a `fallback` key for fn or others, like `key DPAD_UP { ... fn: fallback PAGE_UP }`, then
   500→     * android will first pass an event with original key `DPAD_UP` and {@link KeyEvent#META_FUNCTION_ON}
   501→     * set. But this function will not consume it and android will pass another event with `PAGE_UP`
   502→     * and {@link KeyEvent#META_FUNCTION_ON} not set, which will be consumed.
   503→     *
   504→     * Now there are some other issues as well, firstly ctrl and alt flags are not passed to
   505→     * `getUnicodeChar()`, so modified key values in kcm are not used. Secondly, if the kcm file
   506→     * for other modifiers like shift or fn define a non-alphabet, like { fn: '\u0015' } to act as
   507→     * DPAD_LEFT, the `getUnicodeChar()` will correctly return `21` as the code point but action will
   508→     * not happen because the `handleKeyCode()` function that transforms DPAD_LEFT to `\033[D`
   509→     * escape sequence for the terminal to perform the left action would not be called since its

   566→    @Override
   567→    public boolean onKeyDown(int keyCode, KeyEvent event) {
   568→        if (TERMINAL_VIEW_KEY_LOGGING_ENABLED)
   569→            mClient.logInfo(LOG_TAG, "onKeyDown(keyCode=" + keyCode + ", isSystem()=" + event.isSystem() + ", event=" + event + ")");
   570→        if (mEmulator == null) return true;
   571→        if (isSelectingText()) {
   572→            stopTextSelectionMode();
   573→        }
   574→
   575→        if (mClient.onKeyDown(keyCode, event, mTermSession)) {
   576→            invalidate();
   577→            return true;
   578→        } else if (event.isSystem() && (!mClient.shouldBackButtonBeMappedToEscape() || keyCode != KeyEvent.KEYCODE_BACK)) {
   579→            return super.onKeyDown(keyCode, event);
   580→        } else if (event.getAction() == KeyEvent.ACTION_MULTIPLE && keyCode == KeyEvent.KEYCODE_UNKNOWN) {
   581→            mTermSession.write(event.getCharacters());
   582→            return true;
   583→        } else if (keyCode == KeyEvent.KEYCODE_LANGUAGE_SWITCH) {
   584→            return super.onKeyDown(keyCode, event);
   585→        }
   586→
   587→        final int metaState = event.getMetaState();
   588→        final boolean controlDown = event.isCtrlPressed() || mClient.readControlKey();
   589→        final boolean leftAltDown = (metaState & KeyEvent.META_ALT_LEFT_ON) != 0 || mClient.readAltKey();
   590→        final boolean shiftDown = event.isShiftPressed() || mClient.readShiftKey();
   591→        final boolean rightAltDownFromEvent = (metaState & KeyEvent.META_ALT_RIGHT_ON) != 0;
   592→
   593→        int keyMod = 0;
   594→        if (controlDown) keyMod |= KeyHandler.KEYMOD_CTRL;
   595→        if (event.isAltPressed() || leftAltDown) keyMod |= KeyHandler.KEYMOD_ALT;
   596→        if (shiftDown) keyMod |= KeyHandler.KEYMOD_SHIFT;
   597→        if (event.isNumLockOn()) keyMod |= KeyHandler.KEYMOD_NUM_LOCK;
   598→        // https://github.com/termux/termux-app/issues/731
   599→        if (!event.isFunctionPressed() && handleKeyCode(keyCode, keyMod)) {
   600→            if (TERMINAL_VIEW_KEY_LOGGING_ENABLED) mClient.logInfo(LOG_TAG, "handleKeyCode() took key event");
   601→            return true;
   602→        }
   603→
   604→        // Clear Ctrl since we handle that ourselves:
   605→        int bitsToClear = KeyEvent.META_CTRL_MASK;
   606→        if (rightAltDownFromEvent) {
   607→            // Let right Alt/Alt Gr be used to compose characters.
   608→        } else {
   609→            // Use left alt to send to terminal (e.g. Left Alt+B to jump back a word), so remove:
   610→            bitsToClear |= KeyEvent.META_ALT_ON | KeyEvent.META_ALT_LEFT_ON;
   611→        }
   612→        int effectiveMetaState = event.getMetaState() & ~bitsToClear;
   613→
   614→        if (shiftDown) effectiveMetaState |= KeyEvent.META_SHIFT_ON | KeyEvent.META_SHIFT_LEFT_ON;
   615→        if (mClient.readFnKey()) effectiveMetaState |= KeyEvent.META_FUNCTION_ON;
   616→
   617→        int result = event.getUnicodeChar(effectiveMetaState);
   618→        if (TERMINAL_VIEW_KEY_LOGGING_ENABLED)
   619→            mClient.logInfo(LOG_TAG, "KeyEvent#getUnicodeChar(" + effectiveMetaState + ") returned: " + result);
   620→        if (result == 0) {
   621→            return false;
   622→        }
   623→
   624→        int oldCombiningAccent = mCombiningAccent;
   625→        if ((result & KeyCharacterMap.COMBINING_ACCENT) != 0) {
   626→            // If entered combining accent previously, write it out:
   627→            if (mCombiningAccent != 0)
   628→                inputCodePoint(event.getDeviceId(), mCombiningAccent, controlDown, leftAltDown);
   629→            mCombiningAccent = result & KeyCharacterMap.COMBINING_ACCENT_MASK;
   630→        } else {
   631→            if (mCombiningAccent != 0) {
   632→                int combinedChar = KeyCharacterMap.getDeadChar(mCombiningAccent, result);
   633→                if (combinedChar > 0) result = combinedChar;
   634→                mCombiningAccent = 0;
   635→            }
   636→            inputCodePoint(event.getDeviceId(), result, controlDown, leftAltDown);
   637→        }
   638→
   639→        if (mCombiningAccent != oldCombiningAccent) invalidate();
   640→
   641→        return true;
   642→    }

   644→    public void inputCodePoint(int eventSource, int codePoint, boolean controlDownFromEvent, boolean leftAltDownFromEvent) {
   645→        if (TERMINAL_VIEW_KEY_LOGGING_ENABLED) {
   646→            mClient.logInfo(LOG_TAG, "inputCodePoint(eventSource=" + eventSource + ", codePoint=" + codePoint + ", controlDownFromEvent=" + controlDownFromEvent + ", leftAltDownFromEvent="
   647→                + leftAltDownFromEvent + ")");
   648→        }
   649→
   650→        if (mTermSession == null) return;
   651→
   652→        // Ensure cursor is shown when a key is pressed down like long hold on (arrow) keys
   653→        if (mEmulator != null)
   654→            mEmulator.setCursorBlinkState(true);
   655→
   656→        final boolean controlDown = controlDownFromEvent || mClient.readControlKey();
   657→        final boolean altDown = leftAltDownFromEvent || mClient.readAltKey();
   658→
   659→        if (mClient.onCodePoint(codePoint, controlDown, mTermSession)) return;
   660→
   661→        if (controlDown) {
   662→            if (codePoint >= 'a' && codePoint = 'A' && codePoint  -1) {
   663→            // If not virtual or soft keyboard.
   664→            if (eventSource > KEY_EVENT_SOURCE_SOFT_KEYBOARD) {
   665→                // Work around bluetooth keyboards sending funny unicode characters instead
   666→                // of the more normal ones from ASCII that terminal programs expect - the
   667→                // desire to input the original characters should be low.
   668→                switch (codePoint) {
   669→                    case 0x02DC: // SMALL TILDE.
   670→                        codePoint = 0x007E; // TILDE (~).
   671→                        break;
   672→                    case 0x02CB: // MODIFIER LETTER GRAVE ACCENT.
   673→                        codePoint = 0x0060; // GRAVE ACCENT (`).
   674→                        break;
   675→                    case 0x02C6: // MODIFIER LETTER CIRCUMFLEX ACCENT.
   676→                        codePoint = 0x005E; // CIRCUMFLEX ACCENT (^).
   677→                        break;
   678→                }
   679→            }
   680→
   681→            // If left alt, send escape before the code point to make e.g. Alt+B and Alt+F work in readline:
   682→            mTermSession.writeCodePoint(altDown, codePoint);
   683→        }
   684→    }

   686→    /** Input the specified keyCode if applicable and return if the input was consumed. */
   687→    public boolean handleKeyCode(int keyCode, int keyMod) {
   688→        // Ensure cursor is shown when a key is pressed down like long hold on (arrow) keys
   689→        if (mEmulator != null)
   690→            mEmulator.setCursorBlinkState(true);
   691→
   692→        if (handleKeyCodeAction(keyCode, keyMod))
   693→            return true;
   694→
   695→        TerminalEmulator term = mTermSession.getEmulator();
   696→        String code = KeyHandler.getCode(keyCode, keyMod, term.isCursorKeysApplicationMode(), term.isKeypadApplicationMode());
   697→        if (code == null) return false;
   698→        mTermSession.write(code);
   699→        return true;
   700→    }
   701→
   702→    public boolean handleKeyCodeAction(int keyCode, int keyMod) {
   703→        boolean shiftDown = (keyMod & KeyHandler.KEYMOD_SHIFT) != 0;
   704→
   705→        switch (keyCode) {
   706→            case KeyEvent.KEYCODE_PAGE_UP:
   707→            case KeyEvent.KEYCODE_PAGE_DOWN:
   708→                // shift+page_up and shift+page_down should scroll scrollback history instead of
   709→                // scrolling command history or changing pages
   710→                if (shiftDown) {
   711→                    long time = SystemClock.uptimeMillis();
   712→                    MotionEvent motionEvent = MotionEvent.obtain(time, time, MotionEvent.ACTION_DOWN, 0, 0, 0);
   713→                    doScroll(motionEvent, keyCode == KeyEvent.KEYCODE_PAGE_UP ? -mEmulator.mRows : mEmulator.mRows);
   714→                    motionEvent.recycle();
   715→                    return true;
   716→                }
   717→        }
   718→
   719→       return false;
   720→    }

   759→    public void updateSize() {
   760→        int viewWidth = getWidth();
   761→        int viewHeight = getHeight();
   762→        if (viewWidth == 0 || viewHeight == 0 || mTermSession == null) return;
   763→
   764→        // Set to 80 and 24 if you want to enable vttest.
   765→        int newColumns = Math.max(4, (int) (viewWidth / mRenderer.mFontWidth));
   766→        int newRows = Math.max(4, (viewHeight - mRenderer.mFontLineSpacingAndAscent) / mRenderer.mFontLineSpacing);
   767→
   768→        if (mEmulator == null || (newColumns != mEmulator.mColumns || newRows != mEmulator.mRows)) {
   769→            mTermSession.updateSize(newColumns, newRows, (int) mRenderer.getFontWidth(), mRenderer.getFontLineSpacing());
   770→            mEmulator = mTermSession.getEmulator();
   771→            mClient.onEmulatorSet();
   772→
   773→            // Update mTerminalCursorBlinkerRunnable inner class mEmulator on session change
   774→            if (mTerminalCursorBlinkerRunnable != null)
   775→                mTerminalCursorBlinkerRunnable.setEmulator(mEmulator);
   776→
   777→            mTopRow = 0;
   778→            scrollTo(0, 0);
   779→            invalidate();
   780→        }
   781→    }

   913→    /**
   914→     * Sets whether cursor blinker should be started or stopped. Cursor blinker will only be
   915→     * started if {@link #mTerminalCursorBlinkerRate} does not equal 0 and is between
   916→     * {@link #TERMINAL_CURSOR_BLINK_RATE_MIN} and {@link #TERMINAL_CURSOR_BLINK_RATE_MAX}.
   917→     *
   918→     * This should be called when the view holding this activity is resumed or stopped so that
   919→     * cursor blinker does not run when activity is not visible. If you call this on onResume()
   920→     * to start cursor blinking, then ensure that {@link #mEmulator} is set, otherwise wait for the
   921→     * {@link TerminalViewClient#onEmulatorSet()} event after calling {@link #attachSession(TerminalSession)}
   922→     * for the first session added in the activity since blinking will not start if {@link #mEmulator}
   923→     * is not set, like if activity is started again after exiting it with double back press. Do not
   924→     * call this directly after {@link #attachSession(TerminalSession)} since {@link #updateSize()}
   925→     * may return without setting {@link #mEmulator} since width/height may be 0. Its called again in
   926→     * {@link #onSizeChanged(int, int, int, int)}. Calling on onResume() if emulator is already set
   927→     * is necessary, since onEmulatorSet() may not be called after activity is started after device
   928→     * display timeout with double tap and not power button.
   929→     *
   930→     * It should also be called on the
   931→     * {@link com.termux.terminal.TerminalSessionClient#onTerminalCursorStateChange(boolean)}
   932→     * callback when cursor is enabled or disabled so that blinker is disabled if cursor is not
   933→     * to be shown. It should also be checked if activity is visible if blinker is to be started
   934→     * before calling this.
   935→     *
   936→     * It should also be called after terminal is reset with {@link TerminalSession#reset()} in case
   937→     * cursor blinker was disabled before reset due to call to
   938→     * {@link com.termux.terminal.TerminalSessionClient#onTerminalCursorStateChange(boolean)}.
   939→     *
   940→     * How cursor blinker starting works is by registering a {@link Runnable} with the looper of
   941→     * the main thread of the app which when run, toggles the cursor blinking state and re-registers
   942→     * itself to be called with the delay set by {@link #mTerminalCursorBlinkerRate}. When cursor
   943→     * blinking needs to be disabled, we just cancel any callbacks registered. We don't run our own
   944→     * "thread" and let the thread for the main looper do the work for us, whose usage is also
   945→     * required to update the UI, since it also handles other calls to update the UI as well based
   946→     * on a queue.
   947→     *
   948→     * Note that when moving cursor in text editors like nano, the cursor state is quickly
   949→     * toggled `-> off -> on`, which would call this very quickly sequentially. So that if cursor
   950→     * is moved 2 or more times quickly, like long hold on arrow keys, it would trigger
   951→     * `-> off -> on -> off -> on -> ...`, and the "on" callback at index 2 is automatically
   952→     * cancelled by next "off" callback at index 3 before getting a chance to be run. For this case
   953→     * we log only if {@link #TERMINAL_VIEW_KEY_LOGGING_ENABLED} is enabled, otherwise would clutter
   954→     * the log. We don't start the blinking with a delay to immediately show cursor in case it was
   955→     * previously not visible.
   956→     *
   957→     * @param start If cursor blinker should be started or stopped.
   958→     * @param startOnlyIfCursorEnabled If set to {@code true}, then it will also be checked if the
   959→     *                                 cursor is even enabled by {@link TerminalEmulator} before
   960→     *                                 starting the cursor blinker.
   961→     */
   962→    public synchronized void setTerminalCursorBlinkerState(boolean start, boolean startOnlyIfCursorEnabled) {
   963→        // Stop any existing cursor blinker callbacks
   964→        stopTerminalCursorBlinker();
   965→
   966→        if (mEmulator == null) return;
   967→
   968→        mEmulator.setCursorBlinkingEnabled(false);
   969→
   970→        if (start) {
   971→            // If cursor blinker is not enabled or is not valid
   972→            if (mTerminalCursorBlinkerRate  TERMINAL_CURSOR_BLINK_RATE_MAX)
   973→                return;
   974→            // If cursor blinder is to be started only if cursor is enabled
   975→            else if (startOnlyIfCursorEnabled && ! mEmulator.isCursorEnabled()) {
   976→                if (TERMINAL_VIEW_KEY_LOGGING_ENABLED)
   977→                    mClient.logVerbose(LOG_TAG, "Ignoring call to start cursor blinker since cursor is not enabled");
   978→                return;
   979→            }
   980→
   981→            // Start cursor blinker runnable
   982→            if (TERMINAL_VIEW_KEY_LOGGING_ENABLED)
   983→                mClient.logVerbose(LOG_TAG, "Starting cursor blinker with the blink rate " + mTerminalCursorBlinkerRate);
   984→            if (mTerminalCursorBlinkerHandler == null)
   985→                mTerminalCursorBlinkerHandler = new Handler(Looper.getMainLooper());
   986→            mTerminalCursorBlinkerRunnable = new TerminalCursorBlinkerRunnable(mEmulator, mTerminalCursorBlinkerRate);
   987→            mEmulator.setCursorBlinkingEnabled(true);
   988→            mTerminalCursorBlinkerRunnable.run();
   989→        }
   990→    }
   991→
   992→    /**
   993→     * Cancel the terminal cursor blinker callbacks
   994→     */
   995→    private void stopTerminalCursorBlinker() {
   996→        if (mTerminalCursorBlinkerHandler != null && mTerminalCursorBlinkerRunnable != null) {
   997→            if (TERMINAL_VIEW_KEY_LOGGING_ENABLED)
   998→                mClient.logVerbose(LOG_TAG, "Stopping cursor blinker");
   999→            mTerminalCursorBlinkerHandler.removeCallbacks(mTerminalCursorBlinkerRunnable);
  1000→        }
  1001→    }
  1002→
  1003→    private class TerminalCursorBlinkerRunnable implements Runnable {
  1004→
  1005→        private TerminalEmulator mEmulator;
  1006→        private final int mBlinkRate;
  1007→
  1008→        // Initialize with false so that initial blink state is visible after toggling
  1009→        boolean mCursorVisible = false;
  1010→
  1011→        public TerminalCursorBlinkerRunnable(TerminalEmulator emulator, int blinkRate) {
  1012→            mEmulator = emulator;
  1013→            mBlinkRate = blinkRate;
  1014→        }
  1015→
  1016→        public void setEmulator(TerminalEmulator emulator) {
  1017→            mEmulator = emulator;
  1018→        }
  1019→
  1020→        public void run() {
  1021→            try {
  1022→                if (mEmulator != null) {
  1023→                    // Toggle the blink state and then invalidate() the view so
  1024→                    // that onDraw() is called, which then calls TerminalRenderer.render()
  1025→                    // which checks with TerminalEmulator.shouldCursorBeVisible() to decide whether
  1026→                    // to draw the cursor or not
  1027→                    mCursorVisible = !mCursorVisible;
  1028→                    //mClient.logVerbose(LOG_TAG, "Toggling cursor blink state to " + mCursorVisible);
  1029→                    mEmulator.setCursorBlinkState(mCursorVisible);
  1030→                    invalidate();
  1031→                }
  1032→            } finally {
  1033→                // Recall the Runnable after mBlinkRate milliseconds to toggle the blink state
  1034→                mTerminalCursorBlinkerHandler.postDelayed(this, mBlinkRate);
  1035→            }
  1036→        }
  1037→    }

  1186→                case MotionEvent.ACTION_MOVE:
  1187→                    hideFloatingToolbar();
