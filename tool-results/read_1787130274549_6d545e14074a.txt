   673→   673→      const range = document.createRange();
   674→   674→      let charCount = 0;
   675→   675→      const walker = document.createTreeWalker(display, NodeFilter.SHOW_TEXT);
   676→   676→      let node = walker.nextNode();
   677→   677→      while (node && charCount + node.textContent.length <= nextIdx) {
   678→   678→        charCount += node.textContent.length;
   679→   679→        node = walker.nextNode();
   680→   680→      }
   681→   681→      if (node) {
   682→   682→        const start = nextIdx - charCount;
   683→   683→        range.setStart(node, Math.min(start, node.textContent.length));
   684→   684→        range.setEnd(node, Math.min(start + word.length, node.textContent.length));
   685→   685→        sel.removeAllRanges();
   686→   686→        sel.addRange(range);
   687→   687→      }
   688→   688→    }
   689→   689→  },
   690→   690→
   691→   691→  duplicateLine() {

   695→   695→    Editor.content = lines.join('\n');
   696→   696→    Editor.pushUndo(); Editor.render(); Editor.updateGutter();
   697→   697→    Editor.setCursor(Editor.cursorLine + 1, 1);
   698→   698→  },

   748→   748→  formatDocument() {
   749→   749→    // Basic auto-indent
   750→   750→    const lines = Editor.content.split('\n');
   751→   751→    let indentLevel = 0;
   752→   752→    const indentStr = Editor.useSpaces ? ' '.repeat(Editor.tabSize) : '\t';
   753→   753→    const openBraces = new Set(['{', '(', '[']);
   754→   754→    const closeBraces = new Set(['}', ')', ']']);
   755→   755→    const formatted = lines.map(line => {
   756→   756→      const trimmed = line.trim();
   757→   757→      if (trimmed.length === 0) return '';
   758→   758→      for (const ch of trimmed) { if (closeBraces.has(ch)) indentLevel = Math.max(0, indentLevel - 1); break; }
   759→   759→      const result = indentStr.repeat(indentLevel) + trimmed;
   760→   760→      for (const ch of trimmed) { if (openBraces.has(ch)) indentLevel++; break; }
   761→   761→      return result;
   762→   762→    });
   763→   763→    Editor.content = formatted.join('\n');
   764→   764→    Editor.pushUndo(); Editor.render(); Editor.updateGutter();
   765→   765→  },
