   240→  override fun executeCommand(command: Command?) {
   241→    if (isReleased) {
   242→      return
   243→    }
   244→    if (command == null) {
   245→      log.warn("Cannot execute command in editor. Command is null.")
   246→      return
   247→    }
   248→
   249→    log.info(String.format("Executing command '%s' for completion item.", command.title))
   250→    when (command.command) {
   251→      Command.TRIGGER_COMPLETION -> {
   252→        val completion = getComponent(EditorAutoCompletion::class.java)
   253→        completion.requireCompletion()
   254→      }
   255→
   256→      Command.TRIGGER_PARAMETER_HINTS -> signatureHelp()
   257→      Command.FORMAT_CODE -> formatCodeAsync()
   258→    }

   261→  override fun signatureHelp() {
   262→    if (isReleased) {
   263→      return
   264→    }

   266→    val file = this.file ?: return
   267→

   284→      withContext(Dispatchers.Main) {
   285→        showSignatureHelp(help)
   286→      }
   287→    }.logError("signature help request")
   288→  }

   550→  /**
   551→   * Setup the editor language for the given [file].
   552→   *
   553→   * This applies a proper [Language] and the color scheme to the editor.
   554→   */
   555→  open fun setupLanguage(file: File?) {
   556→    if (isReleased) {
   557→      return
   558→    }
   559→    if (file == null) {
   560→      return
   561→    }
   562→
   563→    createLanguage(file) { language ->
   564→      val extension = file.extension
   565→      if (language is TreeSitterLanguage) {
   566→        IDEColorSchemeProvider.readSchemeAsync(context = context, coroutineScope = editorScope,
   567→          type = extension) { scheme ->
   568→          applyTreeSitterLang(language, extension, scheme)
   569→        }
   570→      } else {
   571→        setEditorLanguage(language)
   572→      }
   573→    }
   574→  }

   576→  /**
   577→   * Applies the given [TreeSitterLanguage] and the [color scheme][scheme] for the given [file type][type].
   578→   */
   579→  open fun applyTreeSitterLang(language: TreeSitterLanguage, type: String,
   580→    scheme: SchemeAndroidIDE?) {
   581→    applyTreeSitterLangInternal(language, type, scheme)
   582→  }
   583→
   584→  private fun applyTreeSitterLangInternal(language: TreeSitterLanguage, type: String,
   585→    scheme: SchemeAndroidIDE?) {
   586→    if (isReleased) {
   587→      return
   588→    }
   589→    var finalScheme = if (scheme != null) {
   590→      scheme
   591→    } else {
   592→      log.error("Failed to read current color scheme")
   593→      SchemeAndroidIDE.newInstance(context)
   594→    }
   595→
   596→    if (finalScheme is IDEColorScheme) {
   597→
   598→      language.setupWith(finalScheme)
   599→
   600→      if (finalScheme.getLanguageScheme(type) == null) {
   601→        log.warn("Color scheme does not support file type '{}'", type)
   602→        finalScheme = SchemeAndroidIDE.newInstance(context)
   603→      }
   604→    }
   605→
   606→    if (finalScheme is DynamicColorScheme) {
   607→      finalScheme.apply(context)
   608→    }
   609→
   610→    colorScheme = finalScheme!!
   611→    setEditorLanguage(language)
   612→  }

   614→  private inline fun createLanguage(file: File, crossinline callback: (Language?) -> Unit) {
   615→
   616→    // 1 -> If the given File object does not represent a file, return emtpy language
   617→    if (!file.isFile) {
   618→      return callback(EmptyLanguage())
   619→    }
   620→
   621→    // 2 -> In case a TreeSitterLanguage has been registered for this file type,
   622→    //      Initialize the TreeSitterLanguage asynchronously and then invoke the callback
   623→    if (TreeSitterLanguageProvider.hasTsLanguage(file)) {
   624→
   625→      // lazily create TS languages as they need to read files from assets
   626→      setupTsLanguageJob = editorScope.launch {
   627→        callback(TreeSitterLanguageProvider.forFile(file, context))
   628→      }.also { job ->
   629→        job.invokeOnCompletion { err ->
   630→          if (err != null) {
   631→            log.error("Failed to setup tree sitter language for file: {}", file, err)
   632→          }
   633→
   634→          setupTsLanguageJob = null
   635→        }
   636→      }
   637→
   638→      return
   639→    }
   640→
   641→    // 3 -> Check if we have ANTLR4 lexer-based languages for this file,
   642→    //      return the language if we do, otherwise return an empty language
   643→    val lang = when (FileUtils.getFileExtension(file)) {
   644→      "gradle" -> GroovyLanguage()
   645→      "c", "h", "cc", "cpp", "cxx" -> CppLanguage()
   646→      else -> EmptyLanguage()
   647→    }
   648→

   658→

   661→    }

   681→      file ?: return@subscribeEvent
   682→
   683→      editorScope.launch {
   684→        dispatchDocumentChangeEvent(event)
   685→        checkForSignatureHelp(event)

   695→        _diagnosticWindow?.dismiss()

   698→      selectionChangeRunner?.also {
   699→        selectionChangeHandler.removeCallbacks(it)
   700→        selectionChangeHandler.postDelayed(it, SELECTION_CHANGE_DELAY)
   701→      }
   702→    }
   703→
   704→    EventBus.getDefault().register(this)
   705→  }

   718→  protected open suspend fun onFindDefinitionResult(
   719→    result: DefinitionResult?,
   720→  ) = withContext(Dispatchers.Main) {
   721→
   722→    if (isReleased) {
   723→      return@withContext
   724→    }
   725→
   726→    val languageClient = languageClient ?: run {
   727→      log.error("No language client found to handle the definitions result")
   728→      return@withContext
   729→    }
   730→
   731→    if (result == null) {
   732→      log.error("Invalid definitions result from language server (null)")
   733→      flashError(string.msg_no_definition)
   734→      return@withContext
   735→    }
   736→
   737→    val locations = result.locations
   738→    if (locations.isEmpty()) {
   739→      log.error("No definitions found")
   740→      flashError(string.msg_no_definition)
   741→      return@withContext
   742→    }
   743→
   744→    if (locations.size != 1) {
   745→      languageClient.showLocations(locations)
   746→      return@withContext
   747→    }
   748→
   749→    val (file1, range) = locations[0]
   750→    if (DocumentUtils.isSameFile(file1, file!!.toPath())) {
   751→      setSelection(range)
   752→      return@withContext
   753→    }
   754→
   755→    languageClient.showDocument(ShowDocumentParams(file1, range))
   756→  }

   758→  protected open suspend fun onFindReferencesResult(result: ReferenceResult?) =
   759→    withContext(Dispatchers.Main) {
   760→
   761→      if (isReleased) {
   762→        return@withContext
   763→      }
   764→
   765→      val languageClient = languageClient ?: run {
   766→        log.error("No language client found to handle the references result")
   767→        return@withContext
   768→      }
   769→
   770→      if (result == null) {
   771→        log.error("Invalid references result from language server (null)")
   772→        flashError(string.msg_no_references)
   773→        return@withContext
   774→      }
   775→
   776→      val locations = result.locations
   777→      if (locations.isEmpty()) {
   778→        log.error("No references found")
   779→        flashError(string.msg_no_references)
   780→        return@withContext
   781→      }
   782→
   783→      if (result.locations.size == 1) {
   784→        val (file, range) = result.locations[0]

   786→        if (DocumentUtils.isSameFile(file, getFile()!!.toPath())) {
   787→          setSelection(range)
   788→          return@withContext
   789→        }
   790→      }
   791→
   792→      languageClient.showLocations(result.locations)
   793→    }

   795→  protected open fun dispatchDocumentOpenEvent() {
   796→    if (isReleased) {
   797→      return
   798→    }
   799→
   800→    val file = this.file ?: return
   801→
   802→    this.fileVersion = 0
   803→
   804→    val openEvent = DocumentOpenEvent(file.toPath(), text.toString(), fileVersion)
   805→
   806→    eventDispatcher.dispatch(openEvent)
   807→  }

   809→  protected open fun dispatchDocumentChangeEvent(event: ContentChangeEvent) {
   810→    if (isReleased) {
   811→      return
   812→    }
   813→
   814→    val file = file?.toPath() ?: return
   815→    var type = ChangeType.INSERT
   816→    if (event.action == ContentChangeEvent.ACTION_DELETE) {
   817→      type = ChangeType.DELETE
   818→    } else if (event.action == ContentChangeEvent.ACTION_SET_NEW_TEXT) {
   819→      type = ChangeType.NEW_TEXT
   820→    }
   821→    var changeDelta = if (type == ChangeType.NEW_TEXT) 0 else event.changedText.length
   822→    if (type == ChangeType.DELETE) {
   823→      changeDelta = -changeDelta
   824→    }
   825→    val start = event.changeStart
   826→    val end = event.changeEnd
   827→    val changeRange = Range(Position(start.line, start.column, start.index),
   828→      Position(end.line, end.column, end.index))
   829→    val changedText = event.changedText.toString()
   830→    val changeEvent = DocumentChangeEvent(file, changedText, text.toString(), ++fileVersion, type,
   831→      changeDelta, changeRange)
   832→
   833→    eventDispatcher.dispatch(changeEvent)
   834→  }

   853→  /**
   854→   * Checks if the content change event should trigger signature help. Signature help trigger
   855→   * characters are :
   856→   *
   857→   *
   858→   *  * `'('` (parentheses)
   859→   *  * `','` (comma)
   860→   *
   861→   *
   862→   * @param event The content change event.
   863→   */
   864→  private fun checkForSignatureHelp(event: ContentChangeEvent) {
   865→    if (isReleased) {
   866→      return
   867→    }
   868→    if (languageServer == null) {
   869→      return
   870→    }
   871→    val changeLength = event.changedText.length
   872→    if (event.action != ContentChangeEvent.ACTION_INSERT || changeLength < 1 || changeLength > 2) {
   873→      // change length will be 1 if ',' is inserted
   874→      // changeLength will be 2 as '(' and ')' are inserted at the same time
   875→      return
   876→    }
   877→
   878→    val ch = event.changedText[0]
   879→    if (ch == '(' || ch == ',') {
   880→      signatureHelp()
   881→    }
   882→  }

   885→    cancelChecker: ICancelChecker) {

   893→  private inline fun <T> safeGet(name: String, action: () -> T): T? {
   894→    return try {
   895→      action()
   896→    } catch (err: Throwable) {
   897→      logError(err, name)
   898→      null
   899→    }
   900→  }

   902→  private fun Job.logError(action: String): Job = apply {
   903→    invokeOnCompletion { err -> logError(err, action) }

   906→  private fun logError(err: Throwable?, action: String) {
   907→    err ?: return
   908→    if (CancelChecker.isCancelled(err)) {
   909→      log.warn("{} has been cancelled", action)
   910→    } else {
   911→      log.error("{} failed", action)
   912→    }
   913→  }
