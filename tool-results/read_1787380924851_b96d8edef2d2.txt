   240→   240→  override fun executeCommand(command: Command?) {
   241→   241→    if (isReleased) {
   242→   242→      return
   243→   243→    }
   244→   244→    if (command == null) {
   245→   245→      log.warn("Cannot execute command in editor. Command is null.")
   246→   246→      return
   247→   247→    }
   248→   248→
   249→   249→    log.info(String.format("Executing command '%s' for completion item.", command.title))
   250→   250→    when (command.command) {
   251→   251→      Command.TRIGGER_COMPLETION -> {
   252→   252→        val completion = getComponent(EditorAutoCompletion::class.java)
   253→   253→        completion.requireCompletion()
   254→   254→      }
   255→   255→
   256→   256→      Command.TRIGGER_PARAMETER_HINTS -> signatureHelp()
   257→   257→      Command.FORMAT_CODE -> formatCodeAsync()
   258→   258→    }
   259→   259→  }

   261→   261→  override fun signatureHelp() {
   262→   262→    if (isReleased) {
   263→   263→      return
   264→   264→    }
   265→   265→    val languageServer = this.languageServer ?: return
   266→   266→    val file = this.file ?: return
   267→   267→
   268→   268→    this.languageClient ?: return
   269→   269→
   270→   270→    sigHelpCancelChecker?.also { it.cancel() }
   271→   271→
   272→   272→    val cancelChecker = JobCancelChecker().also {
   273→   273→      this.sigHelpCancelChecker = it
   274→   274→    }
   275→   275→
   276→   276→    editorScope.launch(Dispatchers.Default) {
   277→   277→      cancelChecker.job = coroutineContext[Job]
   278→   278→
   279→   279→      val help = safeGet("signature help request") {
   280→   280→        val params = SignatureHelpParams(file.toPath(), cursorLSPPosition, cancelChecker)
   281→   281→        languageServer.signatureHelp(params)
   282→   282→      }
   283→   283→
   284→   284→      withContext(Dispatchers.Main) {
   285→   285→        showSignatureHelp(help)
   286→   286→      }
   287→   287→    }.logError("signature help request")
   288→   288→  }

   584→   584→  private fun applyTreeSitterLangInternal(language: TreeSitterLanguage, type: String,
   585→   585→    scheme: SchemeAndroidIDE?) {
   586→   586→    if (isReleased) {
   587→   587→      return
   588→   588→    }
   589→   589→    var finalScheme = if (scheme != null) {
   590→   590→      scheme
   591→   591→    } else {
   592→   592→      log.error("Failed to read current color scheme")
   593→   593→      SchemeAndroidIDE.newInstance(context)
   594→   594→    }
   595→   595→
   596→   596→    if (finalScheme is IDEColorScheme) {
   597→   597→
   598→   598→      language.setupWith(finalScheme)
   599→   599→
   600→   600→      if (finalScheme.getLanguageScheme(type) == null) {
   601→   601→        log.warn("Color scheme does not support file type '{}'", type)
   602→   602→        finalScheme = SchemeAndroidIDE.newInstance(context)
   603→   603→      }
   604→   604→    }
   605→   605→
   606→   606→    if (finalScheme is DynamicColorScheme) {
   607→   607→      finalScheme.apply(context)
   608→   608→    }
   609→   609→
   610→   610→    colorScheme = finalScheme!!
   611→   611→    setEditorLanguage(language)
   612→   612→  }

   614→   614→  private inline fun createLanguage(file: File, crossinline callback: (Language?) -> Unit) {
   615→   615→
   616→   616→    // 1 -> If the given File object does not represent a file, return emtpy language
   617→   617→    if (!file.isFile) {
   618→   618→      return callback(EmptyLanguage())
   619→   619→    }
   620→   620→
   621→   621→    // 2 -> In case a TreeSitterLanguage has been registered for this file type,
   622→   622→    //      Initialize the TreeSitterLanguage asynchronously and then invoke the callback
   623→   623→    if (TreeSitterLanguageProvider.hasTsLanguage(file)) {
   624→   624→
   625→   625→      // lazily create TS languages as they need to read files from assets
   626→   626→      setupTsLanguageJob = editorScope.launch {
   627→   627→        callback(TreeSitterLanguageProvider.forFile(file, context))
   628→   628→      }.also { job ->
   629→   629→        job.invokeOnCompletion { err ->
   630→   630→          if (err != null) {
   631→   631→            log.error("Failed to setup tree sitter language for file: {}", file, err)
   632→   632→          }
   633→   633→
   634→   634→          setupTsLanguageJob = null
   635→   635→        }
   636→   636→      }
   637→   637→
   638→   638→      return
   639→   639→    }
   640→   640→
   641→   641→    // 3 -> Check if we have ANTLR4 lexer-based languages for this file,
   642→   642→    //      return the language if we do, otherwise return an empty language
   643→   643→    val lang = when (FileUtils.getFileExtension(file)) {
   644→   644→      "gradle" -> GroovyLanguage()
   645→   645→      "c", "h", "cc", "cpp", "cxx" -> CppLanguage()
   646→   646→      else -> EmptyLanguage()
   647→   647→    }
   648→   648→
   649→   649→    callback(lang)
   650→   650→  }

   718→   718→  protected open suspend fun onFindDefinitionResult(
   719→   719→    result: DefinitionResult?,
   720→   720→  ) = withContext(Dispatchers.Main) {
   721→   721→
   722→   722→    if (isReleased) {
   723→   723→      return@withContext
   724→   724→    }
   725→   725→
   726→   726→    val languageClient = languageClient ?: run {
   727→   727→      log.error("No language client found to handle the definitions result")
   728→   728→      return@withContext
   729→   729→    }
   730→   730→
   731→   731→    if (result == null) {

   741→   741→      return@withContext
   742→   742→    }
   743→   743→
   744→   744→    if (locations.size != 1) {
   745→   745→      languageClient.showLocations(locations)
   746→   746→      return@withContext
   747→   747→    }
   748→   748→
   749→   749→    val (file1, range) = locations[0]
   750→   750→    if (DocumentUtils.isSameFile(file1, file!!.toPath())) {
   751→   751→      setSelection(range)
   752→   752→      return@withContext
   753→   753→    }
   754→   754→
   755→   755→    languageClient.showDocument(ShowDocumentParams(file1, range))
   756→   756→  }

   809→   809→  protected open fun dispatchDocumentChangeEvent(event: ContentChangeEvent) {
   810→   810→    if (isReleased) {
   811→   811→      return
   812→   812→    }
   813→   813→
   814→   814→    val file = file?.toPath() ?: return
   815→   815→    var type = ChangeType.INSERT
   816→   816→    if (event.action == ContentChangeEvent.ACTION_DELETE) {
   817→   817→      type = ChangeType.DELETE
   818→   818→    } else if (event.action == ContentChangeEvent.ACTION_SET_NEW_TEXT) {
   819→   819→      type = ChangeType.NEW_TEXT
   820→   820→    }
   821→   821→    var changeDelta = if (type == ChangeType.NEW_TEXT) 0 else event.changedText.length
   822→   822→    if (type == ChangeType.DELETE) {
   823→   823→      changeDelta = -changeDelta
   824→   824→    }
   825→   825→    val start = event.changeStart
   826→   826→    val end = event.changeEnd
   827→   827→    val changeRange = Range(Position(start.line, start.column, start.index),
   828→   828→      Position(end.line, end.column, end.index))
   829→   829→    val changedText = event.changedText.toString()
   830→   830→    val changeEvent = DocumentChangeEvent(file, changedText, text.toString(), ++fileVersion, type,
   831→   831→      changeDelta, changeRange)
   832→   832→
   833→   833→    eventDispatcher.dispatch(changeEvent)
   834→   834→  }

   853→   853→  /**
   854→   854→   * Checks if the content change event should trigger signature help. Signature help trigger
   855→   855→   * characters are :
   856→   856→   *
   857→   857→   *
   858→   858→   *  * `'('` (parentheses)
   859→   859→   *  * `','` (comma)
   860→   860→   *
   861→   861→   *
   862→   862→   * @param event The content change event.
   863→   863→   */
   864→   864→  private fun checkForSignatureHelp(event: ContentChangeEvent) {
   865→   865→    if (isReleased) {
   866→   866→      return
   867→   867→    }
   868→   868→    if (languageServer == null) {
   869→   869→      return
   870→   870→    }
   871→   871→    val changeLength = event.changedText.length
   872→   872→    if (event.action != ContentChangeEvent.ACTION_INSERT || changeLength < 1 || changeLength > 2) {
   873→   873→      // change length will be 1 if ',' is inserted
   874→   874→      // changeLength will be 2 as '(' and ')' are inserted at the same time
   875→   875→      return
   876→   876→    }
   877→   877→
   878→   878→    val ch = event.changedText[0]
   879→   879→    if (ch == '(' || ch == ',') {
   880→   880→      signatureHelp()
   881→   881→    }
   882→   882→  }
