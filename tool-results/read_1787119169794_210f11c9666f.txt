   445→            } catch (e: Exception) {
   446→                _snackbarEvents.emit("XML generation failed: ${e.message}")
   447→            } finally {
   448→                _isGenerating.value = false
   449→            }
   450→        }
   451→    }
   452→
   453→    fun generateKotlin() {
   454→        viewModelScope.launch {
   455→            _isGenerating.value = true
   456→            try {
   457→                val kotlin = KotlinGenerator().generate(_canvasState.value)
   458→                _generatedKotlin.value = kotlin
   459→                _snackbarEvents.emit("Kotlin code generated successfully")
   460→            } catch (e: Exception) {
   461→                _snackbarEvents.emit("Kotlin generation failed: ${e.message}")
   462→            } finally {
   463→                _isGenerating.value = false
   464→            }
   465→        }
   466→    }
   467→}

   470→// ===================== Serialization =====================
   471→
   472→@kotlinx.serialization.Serializable
   473→private data class PlacedBlockDto(
   474→    val instanceId: String,
   475→    val definitionId: String,
   476→    val x: Float,
   477→    val y: Float,
   478→    val inputValues: Map<String, String>,
   479→)
   480→
   481→@kotlinx.serialization.Serializable
   482→private data class BlockConnectionDto(
   483→    val id: String,
   484→    val sourceBlockId: String,
   485→    val sourcePortName: String,
   486→    val targetBlockId: String,
   487→    val targetPortName: String,
   488→)
   489→
   490→@kotlinx.serialization.Serializable
   491→private data class BlockCanvasDto(
   492→    val blocks: List<PlacedBlockDto>,
   493→    val connections: List<BlockConnectionDto>,
   494→    val projectName: String,
   495→    val lastSaved: Long?,
   496→)
   497→
   498→class BlockCanvasSerializer : kotlinx.serialization.KSerializer<BlockCanvas> {
   499→    private val dtoSerializer = BlockCanvasDto.serializer()
   500→    override val descriptor = dtoSerializer.descriptor
   501→
   502→    override fun serialize(encoder: kotlinx.serialization.encoding.Encoder, value: BlockCanvas) {
   503→        val dto = BlockCanvasDto(
   504→            blocks = value.blocks.map { PlacedBlockDto(it.instanceId, it.definition.id, it.x, it.y, it.inputValues) },
   505→            connections = value.connections.map { BlockConnectionDto(it.id, it.sourceBlockId, it.sourcePortName, it.targetBlockId, it.targetPortName) },
   506→            projectName = value.projectName,
   507→            lastSaved = value.lastSaved,
   508→        )
   509→        encoder.encodeSerializableValue(dtoSerializer, dto)
   510→    }
   511→
   512→    override fun deserialize(decoder: kotlinx.serialization.encoding.Decoder): BlockCanvas {
   513→        val dto = decoder.decodeSerializableValue(dtoSerializer)
   514→        val allDefinitions = BlockPalette.allBlocks.associateBy { it.id }
   515→        return BlockCanvas(
   516→            blocks = dto.blocks.mapNotNull { bDto ->
   517→                val def = allDefinitions[bDto.definitionId] ?: return@mapNotNull null
   518→                PlacedBlock(bDto.instanceId, def, bDto.x, bDto.y, bDto.inputValues.toMutableMap())
   519→            },
   520→            connections = dto.connections.map { BlockConnection(it.id, it.sourceBlockId, it.sourcePortName, it.targetBlockId, it.targetPortName) },
   521→            projectName = dto.projectName,
   522→            lastSaved = dto.lastSaved,
   523→        )
   524→    }
   525→}

   715→// ===================== Canvas View =====================
   716→
   717→@Composable
   718→private fun BlockCanvasView(
   719→    canvas: BlockCanvas,
   720→    selectedBlockId: String?,
   721→    activeTool: EditorTool,
   722→    zoomLevel: Float,
   723→    panOffset: Offset,
   724→    onBlockMoved: (String, Float, Float) -> Unit,
   725→    onMoveCommit: () -> Unit,
   726→    onBlockSelected: (String?) -> Unit,
   727→    onBlockDoubleTapped: (String) -> Unit,
   728→    onConnectionStart: (String, String) -> Unit,
   729→    onConnectionEnd: (String, String) -> Unit,
   730→    onConnectionRemoved: (String) -> Unit,
   731→    onPanChanged: (Offset) -> Unit,
   732→) {
   733→    var draggingBlockId by remember { mutableStateOf<String?>(null) }
   734→    var dragOffset by remember { mutableStateOf(Offset.Zero) }
   735→    var isPanning by remember { mutableStateOf(false) }
   736→    var panStart by remember { mutableStateOf(Offset.Zero) }
   737→    var panStartOffset by remember { mutableStateOf(Offset.Zero) }
   738→    val connectionLines = remember(canvas.connections, canvas.blocks) {
   739→        canvas.connections.mapNotNull { conn ->
   740→            val source = canvas.blocks.find { it.instanceId == conn.sourceBlockId } ?: return@mapNotNull null
   741→                       val target = canvas.blocks.find { it.instanceId == conn.targetBlockId } ?: return@mapNotNull null
   742→            val sourcePort = source.definition.outputs.find { it.name == conn.sourcePortName } ?: return@mapNotNull null
   743→            val targetPort = target.definition.inputs.find { it.name == conn.targetPortName } ?: return@mapNotNull null
   744→            val sourceIndex = source.definition.outputs.indexOf(sourcePort)
   745→            val targetIndex = target.definition.inputs.indexOf(targetPort)
   746→            val blockWidth = 200f
   747→            val blockHeight = 60f + (source.definition.inputs.size + source.definition.outputs.size) * 20f
   748→            val sx = source.x + blockWidth
   749→            val sy = source.y + 30f + sourceIndex * 20f
   750→            val tx = target.x
   751→            val ty = target.y + 30f + targetIndex * 20f
   752→            Triple(conn.id, Offset(sx, sy), Offset(tx, ty))
   753→        }
   754→    }
   755→
   756→    Box(
   757→        modifier = Modifier
   758→            .fillMaxSize()
   759→            .background(Color(0xFF252526))
   760→            .pointerInput(activeTool) {
   761→                detectTapGestures(
   762→                    onTap = { onBlockSelected(null) },
   763→                    onDoubleTap = { onBlockSelected(null) },
   764→                )
   765→            }
   766→            .pointerInput(activeTool) {
   767→                if (activeTool == EditorTool.Pan) {
   768→                    detectDragGestures(
   769→                        onDragStart = {
   770→                            isPanning = true
   771→                            panStart = it
   772→                            panStartOffset = panOffset
   773→                        },
   774→                        onDrag = { change, dragAmount ->
   775→                            if (isPanning) {
   776→                                val newOffset = Offset(
   777→                                    panStartOffset.x + (change.position.x - panStart.x),
   778→                                    panStartOffset.y + (change.position.y - panStart.y),
   779→                                )
   780→                                onPanChanged(newOffset)
   781→                            }
   782→                        },
   783→                        onDragEnd = { isPanning = false },
   784→                    )
   785→                }
   786→            },
   787→    ) {
   788→        Canvas(
   789→            modifier = Modifier.fillMaxSize(),
   790→        ) {
   791→            val scale = zoomLevel
   792→            val translateX = panOffset.x * scale
   793→            val translateY = panOffset.y * scale
   794→
   795→            connectionLines.forEach { (connId, start, end) ->
   796→                val scaledStart = Offset(start.x * scale + translateX, start.y * scale + translateY)
   797→                val scaledEnd = Offset(end.x * scale + translateX, end.y * scale + translateY)
   798→                val cp1 = Offset(scaledStart.x + 80f, scaledStart.y)
   799→                val cp2 = Offset(scaledEnd.x - 80f, scaledEnd.y)
   800→                drawPath(
   801→                    path = androidx.compose.ui.graphics.Path().apply {
   802→                        moveTo(scaledStart.x, scaledStart.y)
   803→                        cubicTo(cp1.x, cp1.y, cp2.x, cp2.y, scaledEnd.x, scaledEnd.y)
   804→                    },
   805→                    color = Color(0xFF60A5FA),
   806→                    style = Stroke(width = 2f),
   807→                )
   808→            }
   809→        }
   810→
   811→        canvas.blocks.forEach { block ->
   812→            val isSelected = block.instanceId == selectedBlockId
   813→            val blockWidth = 200.dp
   814→            val blockHeight = (60 + (block.definition.inputs.size + block.definition.outputs.size) * 20).dp
   815→
   816→            Box(
   817→                modifier = Modifier
   818→                    .offset(
   819→                        x = (block.x * zoomLevel + panOffset.x * zoomLevel).dp,
   820→                        y = (block.y * zoomLevel + panOffset.y * zoomLevel).dp,
   821→                    )
   822→                    .width(blockWidth * zoomLevel)
   823→                    .height(blockHeight * zoomLevel)
   824→                    .clip(RoundedCornerShape(8.dp * zoomLevel))
   825→                    .background(
   826→                        if (isSelected) Color(0xFF3A3A3C)
   827→                        else Color(0xFF2D2D30)
   828→                    )
   829→                    .then(
   830→                        if (isSelected) Modifier.border(
   831→                            2.dp * zoomLevel,
   832→                            Color(0xFF60A5FA),
   833→                            RoundedCornerShape(8.dp * zoomLevel),
   834→                        ) else Modifier
   835→                    )
   836→                    .pointerInput(activeTool, block.instanceId) {
   837→                        detectDragGestures(
   838→                            onDragStart = {
   839→                                if (activeTool == EditorTool.Select) {
   840→                                    draggingBlockId = block.instanceId
   841→                                    onBlockSelected(block.instanceId)
   842→                                }
   843→                            },
   844→                            onDrag = { change, dragAmount ->
   845→                                if (draggingBlockId == block.instanceId && activeTool == EditorTool.Select) {
   846→                                    change.consume()
   847→                                    val newX = block.x + dragAmount.x / zoomLevel
   848→n                                    val newY = block.y + dragAmount.y / zoomLevel
   849→                                    onBlockMoved(block.instanceId, newX, newY)
   850→                                }
   851→                            },
   852→                            onDragEnd = {
   853→                                if (draggingBlockId == block.instanceId) {
   854→                                    draggingBlockId = null
   855→                                    onMoveCommit()
   856→                                }
   857→                            },
   858→                        )
   859→                    }
   860→                    .pointerInput(block.instanceId) {
   861→                        detectTapGestures(
   862→                            onDoubleTap = { onBlockDoubleTapped(block.instanceId) },
   863→                            onTap = { onBlockSelected(block.instanceId) },
   864→                        )
   865→                    },
   866→                contentAlignment = Alignment.Center,
   867→            ) {
   868→                Column(
   869→                    modifier = Modifier.padding(4.dp * zoomLevel),
   870→                    horizontalAlignment = Alignment.Start,
   871→                ) {
   872→                    Text(
   873→                        text = block.definition.name,
   874→                        color = block.definition.color,
   875→                        fontSize = (12.sp * zoomLevel),
   876→                        fontWeight = FontWeight.Bold,
   877→                        maxLines = 1,
   878→                        overflow = TextOverflow.Ellipsis,
   879→                    )
   880→                    Spacer(Modifier.height(2.dp * zoomLevel))
   881→                    block.definition.inputs.forEach { port ->
   882→                        PortIndicator(
   883→                            port = port,
   884→                            zoomLevel = zoomLevel,
   885→                            onClick = {
   886→                                if (activeTool == EditorTool.Connect) {
   887→                                    onConnectionStart(block.instanceId, port.name)
   888→                                }
   889→                            },
   890→                        )
   891→                    }
   892→                    block.definition.outputs.forEach { port ->
   893→                        PortIndicator(
   894→                            port = port,
   895→                            zoomLevel = zoomLevel,
   896→                            onClick = {
   897→                                if (activeTool == EditorTool.Connect) {
   898→                                    onConnectionEnd(block.instanceId, port.name)
   899→                                }
   900→                            },
   901→                        )
   902→                    }
   903→                }
   904→            }
   905→        }
   906→    }
   907→}

  1024→    modifier: Modifier = Modifier,
  1025→) {
  1026→    var selectedCategory by remember { mutableStateOf(BlockCategory.Events) }
  1027→    var searchQuery by remember { mutableStateOf("") }
  1028→    val filteredBlocks = remember(selectedCategory, searchQuery) {
  1029→        val categoryBlocks = BlockPalette.allBlocks.filter { it.category == selectedCategory }
  1030→        if (searchQuery.isBlank()) categoryBlocks
  1031→        else categoryBlocks.filter { it.name.contains(searchQuery, ignoreCase = true) }
  1032→    }
  1033→
  1034→    Column(modifier = modifier) {

  1036→            text = "Blocks",
  1037→            style = MaterialTheme.typography.titleMedium,
  1038→            modifier = Modifier.padding(12.dp),
  1039→            fontWeight = FontWeight.Bold,
  1040→        )
  1041→        OutlinedTextField(
  1042→            value = searchQuery,
  1043→            onValueChange = { searchQuery = it },
  1044→            placeholder = { Text("Search blocks...", fontSize = 12.sp) },
  1045→            modifier = Modifier
  1046→                .fillMaxWidth()
  1047→                .padding(horizontal = 8.dp),
  1048→            textStyle = TextStyle(fontSize = 12.sp),
  1049→            singleLine = true,
  1050→        )
  1051→        Spacer(Modifier.height(8.dp))
  1052→        LazyRow(
  1053→            modifier = Modifier.fillMaxWidth(),
  1054→            horizontalArrangement = Arrangement.spacedBy(4.dp),
  1055→            contentPadding = androidx.compose.foundation.layout.PaddingValues(horizontal = 8.dp),
  1056→        ) {
  1057→            items(BlockCategory.values()) { category ->
  1058→                FilterChip(
  1059→                    label = category.displayName,
  1060→                    isSelected = selectedCategory == category,
  1061→                    onClick = { selectedCategory = category },
  1062→                    color = when (category) {
  1063→                        BlockCategory.Events -> Color(0xFFFFD700)
  1064→                        BlockCategory.Variables -> Color(0xFFFF6B6B)
  1065→                        BlockCategory.Control -> Color(0xFFFF9500)
  1066→                        BlockCategory.Logic -> Color(0xFF5856D6)
  1067→                        BlockCategory.Loops -> Color(0xFF34C759)
  1068→                        BlockCategory.Math -> Color(0xFF007AFF)
  1069→                        BlockCategory.Text -> Color(0xFFAF52DE)
  1070→                        BlockCategory.Lists -> Color(0xFF5AC8FA)
  1071→                        BlockCategory.UI -> Color(0xFF30B0C7)
  1072→                        BlockCategory.Intent -> Color(0xFFFF3B30)
  1073→                        BlockCategory.Sensor -> Color(0xFF8E8E93)
  1074→                    }
  1075→                )

  1077→        }
  1078→        Spacer(Modifier.height(4.dp))
  1079→        LazyColumn(
  1080→            modifier = Modifier.fillMaxSize(),
  1081→            contentPadding = androidx.compose.foundation.layout.PaddingValues(8.dp),

  1085→                PaletteBlockItem(
  1086→                    definition = definition,

  1088→                )
  1089→            }
  1090→        }
  1091→    }
  1092→}

  1095→private fun FilterChip(
  1096→    label: String,
  1097→    isSelected: Boolean,
  1098→    onClick: () -> Unit,
  1099→    color: Color,
  1100→) {
  1101→    Box(

  1121→        )
