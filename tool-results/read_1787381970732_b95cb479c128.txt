    60→        if (file == null || !file.exists()) {
    61→            return 0;
    62→        }
    63→
    64→        if (!file.isDirectory()) {
    65→            return file.length();
    66→        }
    67→
    68→        List<File> dirs = new LinkedList<>();
    69→        dirs.add(file);
    70→
    71→        long result = 0;
    72→        while (!dirs.isEmpty()) {
    73→            File dir = dirs.remove(0);
    74→            if (!dir.exists())
    75→                continue;
    76→            File[] listFiles = dir.listFiles();
    77→            if (listFiles == null)
    78→                continue;
    79→            for (File child : listFiles) {
    80→                if (child.isDirectory()) {
    81→                    dirs.add(child);
    82→                } else {
    83→                    result += child.length();
    84→                }
    85→            }
    86→        }
    87→
    88→        return result;

    96→        String[] units = {"B", "KiB", "MiB", "GiB"};
    97→        float value = size;
    98→        int unitIndex = 0;
    99→
   100→        while (value >= 1024 && unitIndex < units.length - 1) {
   101→            value /= 1024;
   102→            unitIndex++;
   103→        }
   104→
   105→        if (removeZero && (value - (int) value) * 10 == 0) {
   106→            return String.format("%d %s", (int) value, units[unitIndex]);
   107→        } else {
   108→            return String.format("%.1f %s", value, units[unitIndex]);

   233→    public static void copyDirectory(File source, File copyInto) throws IOException {
   234→        if (!source.isDirectory()) {
   235→            File parentFile = copyInto.getParentFile();
   236→            if (parentFile == null || parentFile.exists() || parentFile.mkdirs()) {
   237→                try (FileInputStream fileInputStream = new FileInputStream(source);
   238→                     FileOutputStream fileOutputStream = new FileOutputStream(copyInto)) {
   239→                    byte[] bArr = new byte[2048];
   240→                    while (true) {
   241→                        int read = fileInputStream.read(bArr);
   242→                        if (read <= 0) {
   243→                            return;
   244→                        }
   245→                        fileOutputStream.write(bArr, 0, read);
   246→                    }
   247→                } catch (IOException e) {
   248→                    Log.e("FileUtil", "Error copying file " + source.getAbsolutePath() + " to " + copyInto.getAbsolutePath(), e);
   249→                    throw e;
   250→                }
   251→            } else {
   252→                throw new IOException("Cannot create dir " + parentFile.getAbsolutePath());
   253→            }
   254→        } else if (copyInto.exists() || copyInto.mkdirs()) {
   255→            String[] list = source.list();
   256→            if (list != null) {
   257→                for (String s : list) {
   258→                    copyDirectory(new File(source, s), new File(copyInto, s));
   259→                }
   260→            }
   261→        } else {
   262→            throw new IOException("Cannot create dir " + copyInto.getAbsolutePath());
   263→        }
   264→    }

   418→    public static String convertUriToFilePath(Context context, Uri uri) {
   419→        String path = null;
   420→        if (DocumentsContract.isDocumentUri(context, uri)) {
   421→            if (isExternalStorageDocument(uri)) {
   422→                String docId = DocumentsContract.getDocumentId(uri);
   423→                String[] split = docId.split(":");
   424→                String type = split[0];
   425→
   426→                if ("primary".equalsIgnoreCase(type)) {
   427→                    path = Environment.getExternalStorageDirectory() + "/" + split[1];
   428→                }
   429→            } else if (isDownloadsDocument(uri)) {
   430→                String id = DocumentsContract.getDocumentId(uri);
   431→
   432→                if (!TextUtils.isEmpty(id)) {
   433→                    if (id.startsWith("raw:")) {
   434→                        return id.replaceFirst("raw:", "");
   435→                    }
   436→                }
   437→
   438→                Uri contentUri = ContentUris
   439→                        .withAppendedId(Uri.parse("content://downloads/public_downloads"), Long.parseLong(id));
   440→
   441→                path = getDataColumn(context, contentUri, null, null);
   442→            } else if (isMediaDocument(uri)) {
   443→                String docId = DocumentsContract.getDocumentId(uri);
   444→                String[] split = docId.split(":");
   445→                String type = split[0];
   446→
   447→                Uri contentUri = null;
   448→                if ("image".equals(type)) {
   449→                    contentUri = MediaStore.Images.Media.EXTERNAL_CONTENT_URI;
   450→                } else if ("video".equals(type)) {
   451→                    contentUri = MediaStore.Video.Media.EXTERNAL_CONTENT_URI;
   452→                } else if ("audio".equals(type)) {
   453→                    contentUri = MediaStore.Audio.Media.EXTERNAL_CONTENT_URI;
   454→                }
   455→
   456→                final String selection = "_id=?";
   457→                String[] selectionArgs = {
   458→                        split[1]
   459→                };
   460→
   461→                path = getDataColumn(context, contentUri, selection, selectionArgs);
   462→            }
   463→        } else if (ContentResolver.SCHEME_CONTENT.equalsIgnoreCase(uri.getScheme())) {
   464→            path = getDataColumn(context, uri, null, null);
   465→        } else if (ContentResolver.SCHEME_FILE.equalsIgnoreCase(uri.getScheme())) {
   466→            path = uri.getPath();
   467→        }
   468→
   469→        if (path != null) {
   470→            try {
   471→                return URLDecoder.decode(path, StandardCharsets.UTF_8);
   472→            } catch (Exception e) {
   473→                return null;
   474→            }
   475→        }
   476→        return null;
   477→    }

   481→
   482→        final String column = MediaStore.Images.Media.DATA;
   483→        String[] projection = {
   484→                column

   486→
   487→        try {
   488→            cursor = context.getContentResolver().query(uri, projection, selection, selectionArgs, null);
   489→            if (cursor != null && cursor.moveToFirst()) {
   490→                int column_index = cursor.getColumnIndexOrThrow(column);
   491→                return cursor.getString(column_index);
   492→            }
   493→        } catch (Exception ignored) {
   494→
   495→        } finally {
   496→            if (cursor != null) {
   497→                cursor.close();
   498→            }
   499→        }
   500→        return null;
   501→    }

   540→    public static int calculateInSampleSize(BitmapFactory.Options options, int reqWidth, int reqHeight) {
   541→        int width = options.outWidth;
   542→        int height = options.outHeight;
   543→        int inSampleSize = 1;
   544→
   545→        if (height > reqHeight || width > reqWidth) {
   546→            int halfHeight = height / 2;
   547→            int halfWidth = width / 2;
   548→
   549→            while ((halfHeight / inSampleSize) >= reqHeight && (halfWidth / inSampleSize) >= reqWidth) {
   550→                inSampleSize *= 2;
   551→            }
   552→        }
   553→
   554→        return inSampleSize;
   555→    }

   633→        int cw = w;
   634→        int ch = h;
   635→
   636→        if (w > width)
   637→            cw = width;
   638→
   639→        if (h > height)
   640→            ch = height;
   641→
   642→        Bitmap bitmap = Bitmap.createBitmap(src, x, y, cw, ch);
   643→        saveBitmap(bitmap, destPath);

   733→                case ExifInterface.ORIENTATION_ROTATE_180 -> 180;

   735→                default -> rotate;

   741→        return rotate;

   748→    public static byte[] readFromInputStream(InputStream stream) {
   749→        int available;
   750→
   751→        try {
   752→            available = stream.available();
   753→        } catch (IOException e) {
   754→            available = 0;
   755→        }
   756→
   757→        ByteArrayOutputStream outputStream = new ByteArrayOutputStream();
   758→        byte[] buffer = new byte[available];
   759→
   760→        try {
   761→            for (int len = stream.read(buffer); len != -1; len = stream.read(buffer)) {
   762→                outputStream.write(buffer, 0, len);
   763→            }
   764→        } catch (IOException e) {
   765→            return new byte[0];
   766→        }
   767→
   768→        return outputStream.toByteArray();
   769→    }

   771→    /**
   772→     * Write bytes to a file.
   773→     *
   774→     * @param target The file to write the data to. Note that it'll get created, even parent directories
   775→     * @param data   The data in bytes to write to. {@link FileUtil#readFromInputStream(InputStream)}
   776→     *               for example, reads bytes
   777→     * @throws IOException Thrown when any exception occurs while operating
   778→     */
   779→    public static void writeBytes(File target, byte[] data) throws IOException {
   780→        if (!target.exists()) {
   781→            target.getParentFile().mkdirs();
   782→        }
   783→        BufferedOutputStream outputStream = new BufferedOutputStream(new FileOutputStream(target));
   784→        outputStream.write(data);
   785→        outputStream.flush();
   786→        outputStream.close();
   787→    }

   789→    public static void extractZipTo(ZipInputStream input, String outPath) throws IOException {
   790→        File outDir = new File(outPath);
   791→        if (!outDir.exists()) {
   792→            outDir.mkdirs();
   793→        }
   794→
   795→        ZipEntry entry = input.getNextEntry();
   796→        while (entry != null) {
   797→            String entryPathExtracted = new File(outPath, entry.getName()).getAbsolutePath();
   798→
   799→            if (entry.isDirectory()) {
   800→                new File(entryPathExtracted).mkdirs();
   801→            } else {
   802→                new File(entryPathExtracted).getParentFile().mkdirs();
   803→                writeBytes(new File(entryPathExtracted), readFromInputStream(input));
   804→            }
   805→            input.closeEntry();
   806→            entry = input.getNextEntry();
   807→        }
   808→        input.close();
   809→    }

   811→    /**
   812→     * Asks the user to grant the current app {@link android.Manifest.permission#MANAGE_EXTERNAL_STORAGE}.
   813→     * Will silently ignore cases where a screen to manage that permission doesn't exist, except on
   814→     * devices with an API level of 29 or lower.
   815→     *
   816→     * @throws AssertionError Thrown if the device's API level is 29 or lower
   817→     */
   818→    public static void requestAllFilesAccessPermission(Context context) {
   819→        if (Build.VERSION.SDK_INT > 29) {
   820→            if (!Environment.isExternalStorageManager()) {
   821→                Intent intent = new Intent();
   822→                intent.setAction(Settings.ACTION_MANAGE_APP_ALL_FILES_ACCESS_PERMISSION);
   823→                intent.setData(Uri.parse("package:" + context.getPackageName()));
   824→                intent.setFlags(Intent.FLAG_ACTIVITY_NEW_TASK);
   825→                try {
   826→                    context.startActivity(intent);
   827→                } catch (ActivityNotFoundException e) {
   828→                    Log.e("FileUtil", "Activity to manage apps' all files access permission not found!");
   829→                }
   830→            }
   831→        } else {
   832→            throw new AssertionError("Not on an API level 30 or higher device!");
   833→        }
   834→    }
