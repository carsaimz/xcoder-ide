   309→  highlightMatchingBracket() {
   310→    document.querySelectorAll('.bracket-match').forEach(e => e.classList.remove('bracket-match'));
   311→    const pos = this.cursorCol - 1;
   312→    const line = this.content.split('\n')[this.cursorLine - 1] || '';
   313→    const ch = line[pos - 1];
   314→    const pair = this.bracketPairs[ch];
   315→    if (!pair) return;
   316→    let depth = 0;
   317→    if (ch === '(' || ch === '[' || ch === '{') {

   592→  _performSearch(query, caseSensitive, direction) {
   593→    const content = caseSensitive ? Editor.content : Editor.content.toLowerCase();
   594→    const q = caseSensitive ? query : query.toLowerCase();
   595→    const matches = [];
   596→    let idx = 0;
   597→    while ((idx = content.indexOf(q, idx)) !== -1) {
   598→      const line = content.substring(0, idx).split('\n').length;
   599→      matches.push({ index: idx, line });
   600→      idx += q.length;
   601→    }
   602→    Editor.searchState.matches = matches;
   603→    Editor.searchState.currentMatch = direction > 0 ? 0 : matches.length - 1;
   604→    if (matches.length > 0) {
   605→      const match = matches[Editor.searchState.currentMatch];
   606→      const lineNum = match.line;
   607→      const lineContent = Editor.content.split('\n')[lineNum - 1] || '';
   608→      const colInLine = match.index - Editor.content.split('\n').slice(0, lineNum - 1).join('\n').length;
   609→      Editor.setCursor(lineNum, colInLine + 1);
   610→    }
   611→    document.getElementById('match-count').textContent = matches.length > 0
   612→      ? `${Editor.searchState.currentMatch + 1}/${matches.length}` : 'No results';
   613→  },

   643→    const singleComment = {kotlin:'//',java:'//',javascript:'//',typescript:'//',c_cpp:'//',css:'/* */',python:'#',ruby:'#',bash:'#',yaml:'#',sql:'--',groovy:'//',swift:'//',dart:'//',php:'//',go:'//',rust:'//'};

   647→    } else if (commentChar === '--') {

   649→    } else {
   650→      lines[lineIdx] = trimmed.startsWith('//') ? line.replace('//','') : line.replace(trimmed.substring(0,2),'//'+trimmed.substring(0,2));

   748→  formatDocument() {
   749→    // Basic auto-indent
   750→    const lines = Editor.content.split('\n');
   751→    let indentLevel = 0;
   752→    const indentStr = Editor.useSpaces ? ' '.repeat(Editor.tabSize) : '\t';
   753→    const openBraces = new Set(['{', '(', '[']);
   754→    const closeBraces = new Set(['}', ')', ']']);
   755→    const formatted = lines.map(line => {
   756→      const trimmed = line.trim();
   757→      if (trimmed.length === 0) return '';
   758→      for (const ch of trimmed) { if (closeBraces.has(ch)) indentLevel = Math.max(0, indentLevel - 1); break; }
   759→      const result = indentStr.repeat(indentLevel) + trimmed;
   760→      for (const ch of trimmed) { if (openBraces.has(ch)) indentLevel++; break; }
   761→      return result;
   762→    });
   763→    Editor.content = formatted.join('\n');
   764→    Editor.pushUndo(); Editor.render(); Editor.updateGutter();
   765→  },

   798→  _renderCommands(filter) {
   799→    const items = document.querySelector('#command-palette .items');
   800→    const lf = filter.toLowerCase();
   801→    const filtered = lf ? this._commands.filter(c => c.label.toLowerCase().includes(lf) || c.category.toLowerCase().includes(lf)) : this._commands;
   802→    items.innerHTML = filtered.map((c, i) =>
   803→      `<div class="item ${i === 0 ? 'selected' : ''}" onclick="EditorActions._executeCommand(${this._commands.indexOf(c)})">${c.label}<span class="shortcut">${c.shortcut || ''}</span><span class="cat">${c.category}</span></div>`
   804→    ).join('');
   805→  },
   806→
   807→  _executeCommand(index) {
   808→    const cmd = this._commands[index];
   809→    if (cmd && cmd.action) cmd.action();
   810→    document.getElementById('command-palette').classList.remove('visible');
   811→  },
   812→

   825→    panel.scrollTop = panel.scrollHeight;

   830→  setProblems(problems) {
   831→    const panel = document.getElementById('problems-panel');
   832→    panel.classList.toggle('visible', problems.length > 0);
   833→    panel.innerHTML = problems.map(p =>
   834→      `<div class="problem ${p.severity}"><span>${p.severity === 'error' ? '✕' : '⚠'}</span><span>${p.message}</span><span class="loc">${p.file}:${p.line}</span></div>`
   835→    ).join('');
   836→  },

   838→  // ---- Snippets ----
   839→  insertSnippet(name) {
   840→    const snippets = Editor.snippets[Editor.language] || Editor.snippets['text'] || {};
   841→    const snippet = snippets[name];
   842→    if (!snippet) return;
   843→    const tabstops = snippet.match(/\$(\d+)|\$\{(\d+)(?::([^}]*))?}/g) || [];
   844→    let text = snippet;
   845→    tabstops.forEach(match => {
   846→      const num = match.replace(/[${}:]/g, '');
   847→      text = text.replace(match, num === '0' ? '$CURSOR' : `placeholder${num}`);
   848→    });
   849→    Editor.insertText(text);
   850→  },
